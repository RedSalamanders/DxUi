[CmdletBinding()]param([string]$Packet)
Set-StrictMode -Version Latest
$ErrorActionPreference='Stop'
if(Test-Path -LiteralPath (Join-Path $Packet 'manifest.receipt.txt')){throw 'Never reseal an existing packet.'}
$rows=@(foreach($file in Get-ChildItem -LiteralPath $Packet -Recurse -File|Sort-Object FullName){
 @{path=[IO.Path]::GetRelativePath($Packet,$file.FullName).Replace('\','/');bytes=$file.Length;sha256=(Get-FileHash -LiteralPath $file.FullName).Hash}
})
$manifest=@{version=1;method='Exact byte hashes of every payload, including verification code and README. manifest.receipt.txt is externally bound by manifest.sha256 and the reported seal.';files=$rows}
[IO.File]::WriteAllText((Join-Path $Packet 'manifest.receipt.txt'),($manifest|ConvertTo-Json -Depth 10),[Text.UTF8Encoding]::new($false))
$seal=(Get-FileHash -LiteralPath (Join-Path $Packet 'manifest.receipt.txt')).Hash
[IO.File]::WriteAllText((Join-Path $Packet 'manifest.sha256'),$seal+"`n",[Text.UTF8Encoding]::new($false))
Write-Output $seal
