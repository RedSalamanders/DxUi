[CmdletBinding()]param([string]$Root)
Set-StrictMode -Version Latest
$ErrorActionPreference='Stop'
$work=Join-Path $Root '.build/reports/rich-markdown-blocks-20261018'
$candidate=Join-Path $work 'Seventh'
$completion=Get-Content (Join-Path $candidate 'completion.json') -Raw|ConvertFrom-Json
if($completion.exitCode -ne 0){throw 'Frozen Release candidate did not pass.'}
$inputs=@(Get-Content (Join-Path $candidate 'inputs.json') -Raw|ConvertFrom-Json)
foreach($input in $inputs){if((Get-FileHash -LiteralPath (Join-Path $Root $input.path)).Hash -cne $input.sha256){throw 'Candidate source changed before final qualification.'}}
$final=Join-Path $work 'Final'
if(Test-Path -LiteralPath $final){throw 'Never overwrite a qualification attempt.'}
New-Item -ItemType Directory -Path $final|Out-Null
$reused=Join-Path $candidate 'ReusedRelease';New-Item -ItemType Directory -Path $reused|Out-Null
$expectedSource=$null;$records=@()
foreach($scope in @('Embedded','EditorControls')){
 $path=Join-Path $Root ('.build/reports/'+$scope+'-x64-Release.json')
 $receipt=Get-Content -LiteralPath $path -Raw|ConvertFrom-Json
 if($receipt.exitCode -ne 0 -or $receipt.skips.Count -or $receipt.PSObject.Properties.Name -contains 'tests'){throw 'Not a complete ordinary success.'}
 if((Get-FileHash -LiteralPath $receipt.executable).Hash -cne $receipt.sha256){throw 'Candidate runtime artifact changed.'}
 $performance=Get-Content -LiteralPath $receipt.performanceReport -Raw|ConvertFrom-Json
 if($null -eq $expectedSource){$expectedSource=$performance.sourceFingerprint}
 if($performance.sourceFingerprint -cne $expectedSource){throw 'Candidate source fingerprints differ.'}
 Copy-Item -LiteralPath $path -Destination $reused
 Copy-Item -LiteralPath (Join-Path $Root ('.build/logs/test-'+$scope+'-x64-Release.log')) -Destination $reused
 Copy-Item -LiteralPath $receipt.performanceReport -Destination $reused -Force
 $records+=@{scope=$scope;sha256=$receipt.sha256;receipt=(Split-Path $path -Leaf);performance=(Split-Path $receipt.performanceReport -Leaf)}
}
Copy-Item -LiteralPath (Join-Path $Root '.build/reports/rich-document-raster') -Destination (Join-Path $reused 'Raster') -Recurse
$artifacts=Join-Path $candidate 'ExactArtifacts';New-Item -ItemType Directory -Path $artifacts|Out-Null
foreach($stem in @('DxUi.ControlTests','DxUi.EmbeddedTests')){foreach($extension in @('.exe','.pdb')){Copy-Item -LiteralPath (Join-Path $Root ('.build/x64/Release/'+$stem+$extension)) -Destination $artifacts}}
@{sourceFingerprint=$expectedSource;files=$inputs;reusedRelease=$records;baseCommit=(git -C $Root rev-parse HEAD);startedUtc=[datetime]::UtcNow.ToString('o');
 notes='Frozen M2 qualification; two exact whole-suite Release successes reused, remaining 55 ordinary suites executed. No performance acceptance or native ARM64/physical IME/AT claim.'}|
 ConvertTo-Json -Depth 12|Set-Content (Join-Path $final 'qualified-source.receipt.txt')
$helpers=@('Invoke-Final.ps1','Invoke-Qualification.ps1','Capture-QualifiedArtifacts.ps1','Invoke-ToolingQualification.ps1')
$helperRecords=@(foreach($name in $helpers){$file=Join-Path $work $name;$copy=Join-Path $final ('Helpers/'+$name);New-Item -ItemType Directory -Path (Split-Path $copy) -Force|Out-Null;Copy-Item -LiteralPath $file -Destination $copy;@{name=$name;sha256=(Get-FileHash -LiteralPath $file).Hash}})
$helperRecords|ConvertTo-Json|Set-Content (Join-Path $final 'helpers.receipt.txt')
& (Join-Path $work 'Invoke-Qualification.ps1') -Root $Root -OutputDirectory $final
foreach($helper in $helperRecords){if((Get-FileHash -LiteralPath (Join-Path $work $helper.name)).Hash -cne $helper.sha256){throw 'Qualification helper changed while running.'}}
$release=Join-Path $final 'x64-Release';New-Item -ItemType Directory -Path $release -Force|Out-Null
Get-ChildItem -LiteralPath $reused -File|Copy-Item -Destination $release
Copy-Item -LiteralPath (Join-Path $reused 'Raster') -Destination (Join-Path $release 'Raster') -Recurse
foreach($input in $inputs){if((Get-FileHash -LiteralPath (Join-Path $Root $input.path)).Hash -cne $input.sha256){throw 'Frozen qualification input changed.'}}
foreach($configuration in @('Debug','Release','ASan Debug')){
 $profile=Join-Path $final ('x64-'+$configuration.Replace(' ','-'))
 foreach($scope in @('Foundation','Embedded','Grid','Theme','Control','MenuExitLifetime','NewControls','EditorControls','TextField','MultilineText','ReadOnly','ComboBox','Tree','Tooltip','Rendering','Animation','Accessibility','WindowHost','InteractiveLease')){
  $receipt=Get-Content (Join-Path $profile ($scope+'-x64-'+$configuration+'.json')) -Raw|ConvertFrom-Json
  if($receipt.exitCode -ne 0 -or $receipt.skips.Count){throw 'Ordinary suite did not pass fully.'}
  if((Get-FileHash -LiteralPath $receipt.executable).Hash -cne $receipt.sha256){throw 'Successful artifact hash mismatch.'}
  $performance=Get-Content -LiteralPath $receipt.performanceReport -Raw|ConvertFrom-Json
  if($performance.sourceFingerprint -cne $expectedSource){throw 'Successful source fingerprint mismatch.'}
 }
}
& (Join-Path $work 'Capture-QualifiedArtifacts.ps1') -Root $Root -OutputDirectory (Join-Path $work 'FinalArtifacts') -ExpectedSource $expectedSource
& (Join-Path $work 'Invoke-ToolingQualification.ps1') -Root $Root -OutputDirectory $final
Write-Host 'All 57 ordinary suite obligations, six builds, exact artifacts and full tooling qualification finished.'
