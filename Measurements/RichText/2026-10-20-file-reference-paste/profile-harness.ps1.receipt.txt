[CmdletBinding()]
param(
    [ValidateSet('Build', 'Directed', 'Test')][string]$Action,
    [ValidateSet('x64', 'ARM64')][string]$Platform = 'x64',
    [ValidateSet('Debug', 'Release', 'ASan Debug')][string]$Configuration = 'Debug',
    [string]$Iteration = 'Final1'
)
Set-StrictMode -Version Latest
$ErrorActionPreference = 'Stop'
$root = 'Z:/src/DxUi-worktrees/rich-text-core-d2'
Set-Location -LiteralPath $root
$report = Join-Path $root '.build/reports/rich-file-reference-paste-20261020'
$name = $Configuration.Replace(' ', '-').ToLowerInvariant()
$packet = Join-Path $report "$Iteration-$Action-$Platform-$name"
if (Test-Path -LiteralPath $packet) { throw 'Refusing to replace retained evidence' }
New-Item -ItemType Directory -Path $packet | Out-Null
Copy-Item -LiteralPath $PSCommandPath -Destination (Join-Path $packet 'harness.ps1')
$env:TEMP = Join-Path $root ".build/image-qualification-temp/Iteration6/build-$Platform-$name"
$env:TMP = $env:TEMP
$env:MSBUILDDISABLENODEREUSE = '1'
New-Item -ItemType Directory -Path $env:TEMP -Force | Out-Null
@{ action=$Action; platform=$Platform; configuration=$Configuration; TEMP=$env:TEMP; TMP=$env:TMP; MSBUILDDISABLENODEREUSE=$env:MSBUILDDISABLENODEREUSE; baseCommit=(git rev-parse HEAD).Trim() } |
    ConvertTo-Json | Set-Content -LiteralPath (Join-Path $packet 'environment.json')
$paths = @(git diff --name-only HEAD -- src include Tests Samples)
@(foreach ($path in $paths) { @{path=$path;sha256=(Get-FileHash -LiteralPath $path).Hash} }) |
    ConvertTo-Json | Set-Content -LiteralPath (Join-Path $packet 'changed-native-inputs.json')
if ($Action -eq 'Build') {
    & ./build.ps1 -Platform $Platform -Configuration $Configuration *> (Join-Path $packet 'runner.log')
    $code = $LASTEXITCODE
    Set-Content -LiteralPath (Join-Path $packet 'exit-code.txt') -Value $code
    Get-Content -LiteralPath (Join-Path $packet 'runner.log') -Tail 8
    if ($code) { exit $code }
    Import-Module (Join-Path $root 'Tools/PerformancePolicy.psm1') -Force
    Get-PerformanceMeasurementState -Root $root -Platform $Platform -Configuration $Configuration |
        ConvertTo-Json -Depth 24 | Set-Content -LiteralPath (Join-Path $packet 'built-state.json')
} else {
    if ($Platform -ne 'x64') { throw 'Native execution is unavailable for ARM64 on this machine' }
    Import-Module (Join-Path $root 'Tools/PerformancePolicy.psm1') -Force
    $before = Get-PerformanceMeasurementState -Root $root -Platform $Platform -Configuration $Configuration
    $before | ConvertTo-Json -Depth 24 | Set-Content -LiteralPath (Join-Path $packet 'before.json')
    if ($Action -eq 'Directed') {
        & ./test.ps1 -Platform $Platform -Configuration $Configuration -SkipBuild -SkipTooling -Suites EditorControls -Tests 'TestRichTextEditorNativeFileReferencePaste,TestRichTextApplicationImageClipboard' *> (Join-Path $packet 'runner.log')
    } else {
        & ./test.ps1 -Platform $Platform -Configuration $Configuration -Full -SkipBuild -SkipTooling *> (Join-Path $packet 'runner.log')
    }
    $code = $LASTEXITCODE
    if ($Action -eq 'Directed' -and -not $code) {
        & (Join-Path $root ".build/x64/$Configuration/DxUi.EmbeddedTests.exe") --test-rich-file-reference-paste *> (Join-Path $packet 'embedded-directed.log')
        $code = $LASTEXITCODE
    }
    Set-Content -LiteralPath (Join-Path $packet 'exit-code.txt') -Value $code
    $after = Get-PerformanceMeasurementState -Root $root -Platform $Platform -Configuration $Configuration
    $after | ConvertTo-Json -Depth 24 | Set-Content -LiteralPath (Join-Path $packet 'after.json')
    Assert-PerformanceMeasurementState -Before $before -After $after
    $receipts = Join-Path $packet 'RunnerReceipts'
    New-Item -ItemType Directory -Path $receipts | Out-Null
    Get-ChildItem -LiteralPath (Join-Path $root '.build/reports') -File |
        Where-Object { $_.Name -like "*x64-$Configuration*" -or ($Configuration -eq 'ASan Debug' -and $_.Name -eq 'AddressSanitizer-x64.json') } |
        ForEach-Object { Copy-Item -LiteralPath $_.FullName -Destination $receipts }
    Get-ChildItem -LiteralPath (Join-Path $root '.build/logs') -File |
        Where-Object { $_.Name -like "test-*-x64-$Configuration*" } |
        ForEach-Object { Copy-Item -LiteralPath $_.FullName -Destination $receipts }
    Get-Content -LiteralPath (Join-Path $packet 'runner.log') -Tail 24
    if ($code) { exit $code }
}
exit 0
