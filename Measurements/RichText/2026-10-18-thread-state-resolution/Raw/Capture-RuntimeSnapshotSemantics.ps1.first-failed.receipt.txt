[CmdletBinding()]param([string]$OutputDirectory)
Set-StrictMode -Version Latest
$ErrorActionPreference='Stop'
New-Item -ItemType Directory -Path $OutputDirectory -Force|Out-Null
$opcodes=@{}
foreach($field in [Reflection.Emit.OpCodes].GetFields([Reflection.BindingFlags]'Public,Static')){if($field.FieldType -eq [Reflection.Emit.OpCode]){$opcode=$field.GetValue($null);$opcodes[[uint16]([int]$opcode.Value -band 0xffff)]=$opcode}}
function Get-MethodIL([Reflection.MethodInfo]$Method){
 $body=$Method.GetMethodBody();if($null -eq $body){return @('No managed IL body.')}
 $bytes=$body.GetILAsByteArray();$index=0;$lines=[Collections.Generic.List[string]]::new()
 while($index -lt $bytes.Length){
  $offset=$index;$code=[uint16]$bytes[$index++];if($code -eq 0xfe){$code=[uint16](0xfe00 -bor $bytes[$index++])}
  $opcode=$opcodes[$code];$operand='';$size=0
  switch($opcode.OperandType.ToString()){
   'InlineNone'{}
   'ShortInlineI'{$size=1;$operand=[string]$bytes[$index]}
   'ShortInlineVar'{$size=1;$operand=[string]$bytes[$index]}
   'ShortInlineBrTarget'{$size=1;$operand=[string]$bytes[$index]}
   'InlineVar'{$size=2;$operand=[string][BitConverter]::ToUInt16($bytes,$index)}
   'InlineI8'{$size=8;$operand=[string][BitConverter]::ToInt64($bytes,$index)}
   'InlineR'{$size=8;$operand=[string][BitConverter]::ToDouble($bytes,$index)}
   'ShortInlineR'{$size=4;$operand=[string][BitConverter]::ToSingle($bytes,$index)}
   'InlineSwitch'{$count=[BitConverter]::ToInt32($bytes,$index);$size=4+4*$count;$operand='targets='+$count}
   default{
    $size=4;$token=[BitConverter]::ToInt32($bytes,$index);$operand='0x{0:X8}' -f $token
    if($opcode.OperandType.ToString() -in @('InlineField','InlineMethod','InlineType','InlineTok')){
     try{$member=$Method.Module.ResolveMember($token);$operand=$member.DeclaringType.FullName+'::'+$member.ToString()}
     catch [ArgumentException]{$operand+=' (generic context unresolved)'}
    }
   }
  }
  $lines.Add(('IL_{0:X4} {1} {2}' -f $offset,$opcode.Name,$operand));$index+=$size
 }
 if($index -ne $bytes.Length){throw 'IL decoder exceeded method body.'}
 return $lines.ToArray()
}

 }
}
