[CmdletBinding()]
param([Parameter(Mandatory)][string]$Iteration)
Set-StrictMode -Version Latest
$ErrorActionPreference='Stop'
$root='Z:/src/DxUi-worktrees/rich-text-core-d2'
Set-Location -LiteralPath $root
$report=Join-Path $root '.build/reports/rich-image-path-m5g1-20261020'
$packet=Join-Path $report $Iteration
if(Test-Path -LiteralPath $packet){throw 'Refusing to replace retained image evidence'}
New-Item -ItemType Directory -Path $packet|Out-Null
$paths=@(git diff --name-only HEAD;git ls-files --others --exclude-standard)|Sort-Object -Unique
foreach($path in $paths){$destination=Join-Path (Join-Path $packet 'AllInputs') $path
New-Item -ItemType Directory -Force -Path (Split-Path $destination)|Out-Null
Copy-Item -LiteralPath $path -Destination $destination}
git diff --binary HEAD | Set-Content -LiteralPath (Join-Path $packet 'working.patch')
git rev-parse HEAD | Set-Content -LiteralPath (Join-Path $packet 'base-commit.txt')
foreach($name in @('DxUi.lib','DxUi.ControlTests.exe','DxUi.ControlTests.pdb','DxUi.EmbeddedTests.exe','DxUi.EmbeddedTests.pdb','DxUi.FoundationTests.exe')){Copy-Item -LiteralPath (Join-Path $root ".build/x64/Debug/$name") -Destination $packet}
Copy-Item -LiteralPath (Join-Path $report ($Iteration.ToLowerInvariant()+'-build.log')) -Destination (Join-Path $packet 'build.log')
Import-Module (Join-Path $root 'Tools/PerformancePolicy.psm1') -Force
$before=Get-PerformanceMeasurementState -Root $root -Platform x64 -Configuration Debug
$before|ConvertTo-Json -Depth 24|Set-Content -LiteralPath (Join-Path $packet 'before.json')
$results=[System.Collections.Generic.List[object]]::new()
foreach($test in @('TestRichTextEditorNativeImagePathPaste','TestRichTextApplicationImageClipboard','TestRichTextImageClipboardTransport','TestRichSemanticOpaqueSourceRegionGrammarAndCaret','TestRichTextEditorNativeRetainedObjectSourceEditing')){
 & (Join-Path $packet 'DxUi.ControlTests.exe') --suite=EditorControls "--test=$test" --no-activate --test-timeout=180 *> (Join-Path $packet "$test.log")
 $code=$LASTEXITCODE
 $results.Add(@{test=$test;exitCode=$code})
 $results|ConvertTo-Json|Set-Content -LiteralPath (Join-Path $packet 'runs.json')
 Write-Output "$test exit=$code"
 if($code -ne 0){exit 1}
}
& (Join-Path $packet 'DxUi.EmbeddedTests.exe') --test-rich-image-failures *> (Join-Path $packet 'image-allocation-failures.log')
$code=$LASTEXITCODE
$results.Add(@{test='Image allocation failures';exitCode=$code})
$results|ConvertTo-Json|Set-Content -LiteralPath (Join-Path $packet 'runs.json')
Write-Output "image-allocation-failures exit=$code"
if($code -ne 0){exit 1}
foreach($scope in @('EditorControls','Accessibility')){
 & (Join-Path $packet 'DxUi.ControlTests.exe') "--suite=$scope" --no-activate --test-timeout=180 *> (Join-Path $packet "$scope.log")
 $code=$LASTEXITCODE
 $results.Add(@{test="$scope complete";exitCode=$code})
 $results|ConvertTo-Json|Set-Content -LiteralPath (Join-Path $packet 'runs.json')
 Write-Output "$scope exit=$code"
 if($code -ne 0){exit 1}
}
foreach($name in @('Embedded','Foundation')){
 & (Join-Path $packet "DxUi.${name}Tests.exe") *> (Join-Path $packet "$name.log")
 $code=$LASTEXITCODE
 $results.Add(@{test="$name complete";exitCode=$code})
 $results|ConvertTo-Json|Set-Content -LiteralPath (Join-Path $packet 'runs.json')
 Write-Output "$name exit=$code"
 if($code -ne 0){exit 1}
}
$after=Get-PerformanceMeasurementState -Root $root -Platform x64 -Configuration Debug
Assert-PerformanceMeasurementState -Before $before -After $after
$after|ConvertTo-Json -Depth 24|Set-Content -LiteralPath (Join-Path $packet 'after.json')
Get-ChildItem -LiteralPath $packet -Recurse -File | ForEach-Object{@{path=$_.FullName;hash=(Get-FileHash -LiteralPath $_.FullName -Algorithm SHA256).Hash}}|ConvertTo-Json|Set-Content -LiteralPath (Join-Path $packet 'files.sha256.json')
