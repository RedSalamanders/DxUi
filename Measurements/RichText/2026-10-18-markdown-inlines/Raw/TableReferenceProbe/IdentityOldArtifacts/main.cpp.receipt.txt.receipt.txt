#include <DxUi/MarkdownSyntax.h>
#include <Windows.h>
#include <crtdbg.h>
#include <cstdlib>
#include <iostream>
#include <set>
int main()
{
    SetErrorMode(SEM_FAILCRITICALERRORS | SEM_NOGPFAULTERRORBOX);
    _set_abort_behavior(0, _WRITE_ABORT_MSG | _CALL_REPORTFAULT);
    _CrtSetReportMode(_CRT_ASSERT, _CRTDBG_MODE_FILE);
    _CrtSetReportFile(_CRT_ASSERT, _CRTDBG_FILE_STDERR);
    DxUi::RichText::MarkdownParser parser;
    const auto source = L"| a | b | c |\n| - | - | - |\n| x |\n";
    for (unsigned revision = 1; revision <= 2; ++revision)
    {
        parser.Begin(source, revision, DxUi::RichText::MarkdownProfile::Gfm029);
        while (parser.HasPendingPreparation()) parser.Advance(1u);
        std::set<std::uint64_t> identities;
        for (const auto& node : parser.Published()->Nodes())
        {
            if (!identities.insert(node.identity).second)
            {
                std::cerr << "Duplicate identity=" << node.identity << " revision=" << revision << '\n';
                return 23;
            }
        }
        std::cout << "Unique revision=" << revision << " nodes=" << identities.size() << '\n';
    }
}
