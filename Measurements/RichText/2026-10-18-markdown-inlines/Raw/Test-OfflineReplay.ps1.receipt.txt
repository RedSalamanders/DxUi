[CmdletBinding()]param([string]$Root,[string]$Packet,[string]$OutputDirectory)
Set-StrictMode -Version Latest
$ErrorActionPreference='Stop'
if(Test-Path -LiteralPath $OutputDirectory){throw 'Fresh offline replay destination required'}
New-Item -ItemType Directory -Path $OutputDirectory|Out-Null
$copy=Join-Path $OutputDirectory 'Packet'
Copy-Item -LiteralPath $Packet -Destination $copy -Recurse
$seal=(Get-Content -LiteralPath (Join-Path $Packet 'manifest.sha256') -Raw).Trim()
$records=[Collections.Generic.List[object]]::new()
$output=& (Join-Path $copy 'Verify.ps1') -Packet $copy -ExpectedSeal $seal
$output|Set-Content (Join-Path $OutputDirectory 'offline-copy.log')
$records.Add(@{operation='offline-exact-copy';passed=$true;output=$output})
$summary=Get-Content -LiteralPath (Join-Path $copy 'summary.receipt.txt') -Raw|ConvertFrom-Json
$targets=@('Native/EditorControls-x64-Debug.receipt.txt',$summary.rasterPairs[0].actual)
foreach($relative in $targets){
 $file=Join-Path $copy $relative
 $original=Join-Path $Packet $relative
 try{
  if($relative.EndsWith('.png')){$bytes=[IO.File]::ReadAllBytes($file);$index=[int]($bytes.Length/2);$bytes[$index]=$bytes[$index] -bxor 1;[IO.File]::WriteAllBytes($file,$bytes)}
  else{[IO.File]::AppendAllText($file," ",[Text.UTF8Encoding]::new($false))}
  $rejected=$false;$message=''
  try{& (Join-Path $copy 'Verify.ps1') -Packet $copy -ExpectedSeal $seal|Out-Null}
  catch{if($_.Exception.Message -notlike '*Packet file hash/length mismatch:*'){throw};$rejected=$true;$message=$_.Exception.Message}
  if(-not $rejected){throw "Tampered evidence accepted: $relative"}
  $records.Add(@{operation='tamper-rejection';target=$relative;passed=$true;message=$message})
 }finally{Copy-Item -LiteralPath $original -Destination $file}
}
$output=& (Join-Path $copy 'Verify.ps1') -Packet $copy -ExpectedSeal $seal
$output|Set-Content (Join-Path $OutputDirectory 'restored-copy.log')
$output=& (Join-Path $Packet 'Verify.ps1') -Packet $Packet -ExpectedSeal $seal
$output|Set-Content (Join-Path $OutputDirectory 'original-after-tamper.log')
@{version=1;seal=$seal;utc=[datetime]::UtcNow.ToString('o');records=$records.ToArray();originalAndRestoredVerify=$true}|ConvertTo-Json -Depth 10|Set-Content (Join-Path $OutputDirectory 'replay.receipt.txt')
Write-Output 'Exact-copy replay, receipt/PNG tamper rejection and original/restored verification passed.'
