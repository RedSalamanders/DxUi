function(escape_xml value result)
    string(REPLACE "&" "&amp;" value "${value}")
    string(REPLACE "<" "&lt;" value "${value}")
    string(REPLACE ">" "&gt;" value "${value}")
    string(REPLACE "\"" "&quot;" value "${value}")
    set(${result} "${value}" PARENT_SCOPE)
endfunction()
escape_xml("${DXUI_ROOT}" DXUI_ROOT_XML)
escape_xml("${DXUI_LOCK_FILE}" DXUI_LOCK_XML)
escape_xml("${DXUI_CONSUMER_OUTPUT_ROOT}" DXUI_OUTPUT_XML)
