constexpr UINT FileResultMessage = WM_APP + 0x71;
enum Command : UINT
{
    Open = 101,
    Save,
    SaveAs,
    SourceView,
    ReadOnly,
    BoldText,
    ItalicText,
    CodeText,
    UndoEdit,
    RedoEdit
};
const wchar_t *Sample = FeatureShowcase;
struct IoResult
{
    bool open{};
    std::filesystem::path path;
    std::wstring source;
    std::uint64_t revision{};
    std::string error;
};
struct Mailbox
{
    std::mutex mutex;
    std::optional<IoResult> result;
    std::atomic_bool alive{true};
    HWND window{};
};
class Application
{
  public:
    ComPtr<IDWriteFactory> factory;
    std::unique_ptr<NativeControl> editor;
    HWND window{};
    std::filesystem::path path;
    std::shared_ptr<Mailbox> mailbox = std::make_shared<Mailbox>();
    std::jthread io;
    bool ioBusy{};
    std::wstring status;
    std::wstring savedSource = Sample;
    Application()
    {
        status.reserve(1024);
        Check(DWriteCreateFactory(DWRITE_FACTORY_TYPE_SHARED, __uuidof(IDWriteFactory),
                                  reinterpret_cast<IUnknown **>(factory.GetAddressOf())),
              "DWrite factory");
        editor = std::make_unique<NativeControl>(factory.Get());
    }
    ~Application()
    {
        status.reserve(1024);
        mailbox->alive = false;
        if (window && IsWindow(window))
            DestroyWindow(window);
    }
    void Create(HINSTANCE instance, bool visible)
    {
        WNDCLASSEXW type{sizeof(type)};
        type.hInstance = instance;
        type.lpfnWndProc = Procedure;
        type.lpszClassName = L"RichEditRebuild.Application";
        type.hCursor = LoadCursorW(nullptr, IDC_ARROW);
        if (!RegisterClassExW(&type) && GetLastError() != ERROR_CLASS_ALREADY_EXISTS)
            throw std::runtime_error("App class");
        window = CreateWindowExW(0, type.lpszClassName, L"RichEditRebuild", WS_OVERLAPPEDWINDOW, CW_USEDEFAULT,
                                 CW_USEDEFAULT, 1100, 760, nullptr, nullptr, instance, this);
        if (!window)
            throw std::runtime_error("App window");
        mailbox->window = window;
        editor->Create(window, instance, 1, true);
        editor->SetText(Sample);
        HMENU menu = CreateMenu(), file = CreatePopupMenu(), view = CreatePopupMenu(), format = CreatePopupMenu();
        AppendMenuW(file, MF_STRING, Open, L"&Open UTF-8...\tCtrl+O");
        AppendMenuW(file, MF_STRING, Save, L"&Save\tCtrl+S");
        AppendMenuW(file, MF_STRING, SaveAs, L"Save &as...");
        AppendMenuW(view, MF_STRING, SourceView, L"Source / formatted\tF6");
        AppendMenuW(view, MF_STRING, ReadOnly, L"Read-only");
        AppendMenuW(format, MF_STRING, BoldText, L"Bold\tCtrl+B");
        AppendMenuW(format, MF_STRING, ItalicText, L"Italic\tCtrl+I");
        AppendMenuW(format, MF_STRING, CodeText, L"Code");
        AppendMenuW(format, MF_STRING, UndoEdit, L"Undo\tCtrl+Z");
        AppendMenuW(format, MF_STRING, RedoEdit, L"Redo\tCtrl+Y");
        AppendMenuW(menu, MF_POPUP, reinterpret_cast<UINT_PTR>(file), L"&File");
        AppendMenuW(menu, MF_POPUP, reinterpret_cast<UINT_PTR>(view), L"&View");
        AppendMenuW(menu, MF_POPUP, reinterpret_cast<UINT_PTR>(format), L"F&ormat");
        SetMenu(window, menu);
        Arrange();
        Caption();
        if (visible)
        {
            ShowWindow(window, SW_SHOW);
            SetFocus(editor->Window());
        }
    }
    void Arrange()
    {
        if (editor && editor->Window())
        {
            RECT bounds{};
            GetClientRect(window, &bounds);
            MoveWindow(editor->Window(), 0, 0, bounds.right, bounds.bottom, TRUE);
        }
    }
    void Caption()
    {
        const auto caption = L"RichEditRebuild â€” Unicode C++ prototype â€” " +
                             std::wstring(editor->Mode() == ViewMode::Rich ? L"Formatted" : L"Source") +
                             (editor->ReadOnly() ? L" â€” Read-only" : L"") + L" â€” " +
                             (path.empty() ? L"Untitled" : path.filename().wstring()) +
                             (status.empty() ? L"" : L" â€” " + status);
        SetWindowTextW(window, caption.c_str());
    }
    std::optional<std::filesystem::path> Pick(bool save)
    {
        std::array<wchar_t, 32768> buffer{};
        if (save && !path.empty())
            wcsncpy_s(buffer.data(), buffer.size(), path.c_str(), _TRUNCATE);
        OPENFILENAMEW dialog{sizeof(dialog)};
        dialog.hwndOwner = window;
        dialog.lpstrFilter = L"Markdown\0*.md;*.markdown\0All files\0*.*\0";
        dialog.lpstrFile = buffer.data();
        dialog.nMaxFile = static_cast<DWORD>(buffer.size());
        dialog.lpstrDefExt = L"md";
        dialog.Flags = OFN_NOCHANGEDIR | OFN_PATHMUSTEXIST | (save ? OFN_OVERWRITEPROMPT : OFN_FILEMUSTEXIST);
        if (save ? GetSaveFileNameW(&dialog) : GetOpenFileNameW(&dialog))
            return std::filesystem::path(buffer.data());
        return {};
    }
    void FileJob(bool open, std::filesystem::path destination)
    {
        if (ioBusy)
        {
            status = L"A file operation is still running";
            Caption();
            return;
        }
        if (io.joinable())
            io.join();
        ioBusy = true;
        status = open ? L"Opening..." : L"Saving...";
        Caption();
        const auto source = editor->Model().Snapshot();
        const auto revision = editor->Model().Revision();
        const auto box = mailbox;
        io = std::jthread([box, source, revision, destination = std::move(destination), open] {
            IoResult result;
            result.open = open;
            result.path = destination;
            result.revision = revision;
            try
            {
                if (open)
                    result.source = ReadUtf8File(destination);
                else
                {
                    result.source = source.Text();
                    WriteUtf8FileAtomic(destination, result.source);
                }
            }
            catch (const std::exception &error)
            {
                result.error = error.what();
            }
            {
                std::scoped_lock lock(box->mutex);
                box->result = std::move(result);
            }
            if (box->alive)
                PostMessageW(box->window, FileResultMessage, 0, 0);
        });
    }
    void FileCompleted()
    {
        std::optional<IoResult> result;
        {
            std::scoped_lock lock(mailbox->mutex);
            result = std::move(mailbox->result);
            mailbox->result.reset();
        }
        if (!result)
            return;
        ioBusy = false;
        if (!result->error.empty())
            status = DecodeUtf8(result->error);
        else if (result->open && result->revision != editor->Model().Revision())
            status = L"Open cancelled because the document changed";
        else
        {
            path = result->path;
            savedSource = result->source;
            if (result->open)
                editor->SetText(std::move(result->source));
            status = result->open                                     ? L"Opened"
                     : result->revision == editor->Model().Revision() ? L"Saved"
                                                                      : L"Snapshot saved; newer edits remain";
        }
        Caption();
    }
    void Execute(UINT command)
    {
        if (command == SourceView)
            editor->SetMode(editor->Mode() == ViewMode::Rich ? ViewMode::Source : ViewMode::Rich);
        if (command == ReadOnly)
            editor->SetReadOnly(!editor->ReadOnly());
        if (command == BoldText)
            editor->Format(L"**");
        if (command == ItalicText)
            editor->Format(L"*");
        if (command == CodeText)
            editor->Format(L"`");
        if (command == UndoEdit)
            SendMessageW(editor->Window(), WM_UNDO, 0, 0);
        if (command == RedoEdit)
            SendMessageW(editor->Window(), EM_REDO, 0, 0);
        if (command == Open)
        {
            if (editor->GetText() != savedSource &&
                MessageBoxW(window, L"Discard unsaved changes and open another document?", L"RichEditRebuild",
                            MB_OKCANCEL | MB_ICONWARNING | MB_DEFBUTTON2) != IDOK)
                return;
            if (const auto selected = Pick(false))
                FileJob(true, *selected);
        }
        if (command == Save || command == SaveAs)
        {
            if (command == Save && !path.empty())
                FileJob(false, path);
            else if (const auto selected = Pick(true))
                FileJob(false, *selected);
        }
        Caption();
    }
    static LRESULT CALLBACK Procedure(HWND window, UINT message, WPARAM wParam, LPARAM lParam) noexcept
    {
        auto *app = reinterpret_cast<Application *>(GetWindowLongPtrW(window, GWLP_USERDATA));
        if (message == WM_NCCREATE)
        {
            app = static_cast<Application *>(reinterpret_cast<CREATESTRUCTW *>(lParam)->lpCreateParams);
            SetWindowLongPtrW(window, GWLP_USERDATA, reinterpret_cast<LONG_PTR>(app));
            app->window = window;
            return TRUE;
        }
        if (app)
            try
            {
                switch (message)
                {
                case WM_SIZE:
                    app->Arrange();
                    return 0;
                case WM_COMMAND:
                    app->Execute(LOWORD(wParam));
                    return 0;
                case FileResultMessage:
                    app->FileCompleted();
                    return 0;
                case WM_SETFOCUS:
                    if (app->editor->Window())
                        SetFocus(app->editor->Window());
                    return 0;
                case WM_CLOSE:
                    if (app->ioBusy)
                    {
                        app->status = L"Wait for the current file operation before closing";
                        app->Caption();
                        return 0;
                    }
                    if (app->editor->GetText() != app->savedSource &&
                        MessageBoxW(window, L"Discard unsaved changes and close?", L"RichEditRebuild",
                                    MB_YESNO | MB_ICONWARNING | MB_DEFBUTTON2) != IDYES)
                        return 0;
                    DestroyWindow(window);
                    return 0;
                case WM_DESTROY:
                    app->mailbox->alive = false;
                    PostQuitMessage(0);
                    return 0;
                case WM_NCDESTROY:
                    app->window = nullptr;
                    SetWindowLongPtrW(window, GWLP_USERDATA, 0);
                    break;
                case WM_DPICHANGED: {
                    const auto *rect = reinterpret_cast<const RECT *>(lParam);
                    SetWindowPos(window, nullptr, rect->left, rect->top, rect->right - rect->left,
                                 rect->bottom - rect->top, SWP_NOZORDER | SWP_NOACTIVATE);
                    return 0;
                }
                }
            }
            catch (const std::exception &error)
            {
                app->status = L"Operation failed; the document was retained";
                OutputDebugStringA(error.what());
                app->Caption();
                return 0;
            }
        return DefWindowProcW(window, message, wParam, lParam);
    }
    void WaitFile()
    {
        const auto until = GetTickCount64() + 5000;
        while (ioBusy)
        {
            MSG message{};
            while (PeekMessageW(&message, nullptr, 0, 0, PM_REMOVE))
                DispatchMessageW(&message);
            if (GetTickCount64() > until)
                throw std::runtime_error("App I/O timeout");
            MsgWaitForMultipleObjectsEx(0, nullptr, 20, QS_ALLINPUT, MWMO_INPUTAVAILABLE);
        }
    }
    void WaitPrepared()
    {
        const auto until = GetTickCount64() + 5000;
        while (!editor->Prepared())
        {
            MSG message{};
            while (PeekMessageW(&message, nullptr, 0, 0, PM_REMOVE))
                DispatchMessageW(&message);
            if (!editor->Error().empty())
                throw std::runtime_error(editor->Error());
            if (GetTickCount64() > until)
                throw std::runtime_error("App preparation timeout");
            MsgWaitForMultipleObjectsEx(0, nullptr, 20, QS_ALLINPUT, MWMO_INPUTAVAILABLE);
        }
    }
};
} // namespace
int WINAPI wWinMain(HINSTANCE instance, HINSTANCE, PWSTR, int)
{
    const HRESULT apartment = CoInitializeEx(nullptr, COINIT_APARTMENTTHREADED);
    if (FAILED(apartment))
        return 1;
    int result = 0;
    try
    {
        SetProcessDpiAwarenessContext(DPI_AWARENESS_CONTEXT_PER_MONITOR_AWARE_V2);
        int count = 0;
        auto arguments = CommandLineToArgvW(GetCommandLineW(), &count);
        if (!arguments)
            throw std::runtime_error("Command line");
        std::wstring action = count > 1 ? arguments[1] : L"";
        std::filesystem::path output = count > 2 ? arguments[2] : L"";
        LocalFree(arguments);
        Application app;
        const bool capture = action == L"--capture", smoke = action == L"--smoke";
        app.Create(instance, !capture && !smoke);
        if (capture || smoke)
        {
            app.WaitPrepared();
            if (capture)
                app.editor->CapturePng(output, 1000, 640);
            else
            {
                app.editor->SetText(L"# App test\r\n**Unicode** \U0001f600");
                app.WaitPrepared();
                app.editor->Select({0, 0});
                SendMessageW(app.editor->Window(), WM_CHAR, L'X', 0);
                app.WaitPrepared();
                const auto expected = app.editor->GetText();
                app.Execute(SourceView);
                app.WaitPrepared();
                app.Execute(SourceView);
                app.WaitPrepared();
                if (app.editor->GetText() != expected)
                    throw std::runtime_error("App mode switch modified source");
                app.FileJob(false, output);
                app.WaitFile();
                if (ReadUtf8File(output) != expected)
                    throw std::runtime_error("App save/reopen failed");
                app.editor->SetText(L"temporary");
                app.FileJob(true, output);
                app.WaitFile();
                app.WaitPrepared();
                if (app.editor->GetText() != expected)
                    throw std::runtime_error("Async open changed saved content");
                app.FileJob(true, output);
                app.editor->SetText(L"newer edits");
                app.WaitFile();
                app.WaitPrepared();
                if (app.editor->GetText() != L"newer edits")
                    throw std::runtime_error("Stale open clobbered newer edits");
                app.FileJob(true, output.wstring() + L".missing");
                app.WaitFile();
                if (app.editor->GetText() != L"newer edits")
                    throw std::runtime_error("Failed open clobbered source");
                app.editor->SetText(expected);
                app.WaitPrepared();
                app.editor->CapturePng(output.wstring() + L".png", 800, 480);
            }
        }
        else
        {
            if (!action.empty())
            {
                app.path = action;
                app.FileJob(true, app.path);
                app.Caption();
            }
            MSG message{};
            while (GetMessageW(&message, nullptr, 0, 0) > 0)
            {
                if (message.message == WM_KEYDOWN)
                {
                    if (message.wParam == VK_F6)
                    {
                        app.Execute(SourceView);
                        continue;
                    }
                    if ((GetKeyState(VK_CONTROL) & 0x8000) != 0 && (GetKeyState(VK_MENU) & 0x8000) == 0 &&
                        (message.wParam == 'O' || message.wParam == 'S'))
                    {
                        app.Execute(message.wParam == 'O' ? Open : Save);
                        continue;
                    }
                }
                TranslateMessage(&message);
                DispatchMessageW(&message);
            }
        }
    }
    catch (const std::exception &error)
    {
        OutputDebugStringA(error.what());
        result = 1;
    }
    CoUninitialize();
    return result;
}
