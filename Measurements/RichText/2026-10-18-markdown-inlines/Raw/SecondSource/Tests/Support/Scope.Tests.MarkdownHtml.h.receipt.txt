#pragma once
#include <DxUi/MarkdownSyntax.h>
#include <algorithm>
#include <string>
#include <vector>

// Test-only AST serializer. Original expected HTML is generated from byte-pinned
// external specifications, never from this serializer or the owned parser.
// Raw HTML output is deliberately not a safe browser preview/export API.
namespace MarkdownHtmlTests
{
using namespace DxUi::RichText;
inline std::wstring Escape(std::wstring_view value)
{
    std::wstring result;
    for (const auto c : value)
        switch (c)
        {
        case L'&': result += L"&amp;"; break;
        case L'<': result += L"&lt;"; break;
        case L'>': result += L"&gt;"; break;
        case L'"': result += L"&quot;"; break;
        default: result += c ? c : L'\ufffd'; break;
        }
    return result;
}
inline std::wstring Url(std::wstring_view value)
{
    constexpr wchar_t hex[] = L"0123456789ABCDEF";
    std::wstring result;
    const auto byte = [&](unsigned c) { result += L'%'; result += hex[c >> 4u]; result += hex[c & 15u]; };
    for (std::size_t i = 0; i < value.size(); ++i)
    {
        auto c = static_cast<unsigned>(value[i]);
        if (c >= 0xd800u && c <= 0xdbffu && i + 1u < value.size())
        { const auto lead = c; c = 0x10000u + ((lead - 0xd800u) << 10u) + static_cast<unsigned>(value[++i]) - 0xdc00u; }
        if ((c >= L'a' && c <= L'z') || (c >= L'A' && c <= L'Z') || (c >= L'0' && c <= L'9') ||
            (c < 128u && std::wstring_view(L";/?:@&=+$,-_.!~*'()#%").find(static_cast<wchar_t>(c)) != std::wstring_view::npos)) result += static_cast<wchar_t>(c);
        else if (c < 128u) byte(c);
        else if (c < 0x800u) { byte(0xc0u | (c >> 6u)); byte(0x80u | (c & 63u)); }
        else if (c < 0x10000u) { byte(0xe0u | (c >> 12u)); byte(0x80u | ((c >> 6u) & 63u)); byte(0x80u | (c & 63u)); }
        else { byte(0xf0u | (c >> 18u)); byte(0x80u | ((c >> 12u) & 63u)); byte(0x80u | ((c >> 6u) & 63u)); byte(0x80u | (c & 63u)); }
    }
    return Escape(result);
}
inline std::wstring Filter(std::wstring_view raw, MarkdownExtensions extensions)
{
    if (!HasMarkdownExtension(extensions,MarkdownExtensions::TagFilter)) return std::wstring(raw);
    std::wstring result;
    for (std::size_t i = 0; i < raw.size(); ++i)
    {
        if (raw[i] == L'<')
        {
            auto first = i + 1u;
            if (first < raw.size() && raw[first] == L'/') ++first;
            auto end = first;
            while (end < raw.size() && ((raw[end] >= L'a' && raw[end] <= L'z') || (raw[end] >= L'A' && raw[end] <= L'Z'))) ++end;
            std::wstring name(raw.substr(first,end-first));
            for (auto& c : name) if (c >= L'A' && c <= L'Z') c = static_cast<wchar_t>(c + 32);
            constexpr std::wstring_view disallowed[] = {L"title",L"textarea",L"style",L"xmp",L"iframe",L"noembed",L"noframes",L"script",L"plaintext"};
            if (end < raw.size() && std::wstring_view(L" \t\n\r\f/>").find(raw[end]) != std::wstring_view::npos &&
                std::find(std::begin(disallowed),std::end(disallowed),name) != std::end(disallowed))
            { result += L"&lt;"; continue; }
        }
        result += raw[i];
    }
    return result;
}
inline std::wstring Inline(const MarkdownSyntax& syntax, std::size_t first, bool plain = false)
{
    struct Visit { std::size_t index; bool close, plain; };
    std::vector<Visit> pending;
    if (first != MarkdownNoNode) pending.push_back({first,false,plain});
    std::wstring result;
    while (!pending.empty())
    {
        const auto visit = pending.back(); pending.pop_back();
        const auto& node = syntax.Inlines().at(visit.index);
        if (!visit.close && node.nextSibling != MarkdownNoNode) pending.push_back({node.nextSibling,false,visit.plain});
        std::wstring_view tag;
        switch (node.kind)
        {
        case MarkdownInlineKind::Emphasis: tag = L"em"; break;
        case MarkdownInlineKind::Strong: tag = L"strong"; break;
        case MarkdownInlineKind::Strikethrough: tag = L"del"; break;
        case MarkdownInlineKind::Link: tag = L"a"; break;
        default: break;
        }
        if (visit.close) { if (!visit.plain && !tag.empty()) result += L"</" + std::wstring(tag) + L">"; continue; }
        if (node.kind == MarkdownInlineKind::Text) result += visit.plain ? node.value : Escape(node.value);
        else if (node.kind == MarkdownInlineKind::Code) result += visit.plain ? node.value : L"<code>" + Escape(node.value) + L"</code>";
        else if (node.kind == MarkdownInlineKind::SoftBreak) result += L'\n';
        else if (node.kind == MarkdownInlineKind::HardBreak) result += visit.plain ? L"\n" : L"<br />\n";
        else if (node.kind == MarkdownInlineKind::Html) { if (!visit.plain) result += Filter(node.value,syntax.Extensions()); }
        else if (node.kind == MarkdownInlineKind::Image && !visit.plain)
        {
            result += L"<img src=\"" + Url(node.destination) + L"\" alt=\"" + Escape(Inline(syntax,node.firstChild,true)) + L"\"";
            if (!node.title.empty()) result += L" title=\"" + Escape(node.title) + L"\"";
            result += L" />";
        }
        else
        {
            if (!visit.plain && !tag.empty())
            {
                result += L"<" + std::wstring(tag);
                if (node.kind == MarkdownInlineKind::Link)
                { result += L" href=\"" + Url(node.destination) + L"\""; if (!node.title.empty()) result += L" title=\"" + Escape(node.title) + L"\""; }
                result += L">";
            }
            pending.push_back({visit.index,true,visit.plain});
            if (node.firstChild != MarkdownNoNode) pending.push_back({node.firstChild,false,visit.plain});
        }
    }
    return result;
}
inline std::wstring Lines(const MarkdownSyntax& syntax, const MarkdownNode& node)
{
    std::wstring result;
    for (const auto& line : node.lines)
    {
        result.append(line.leadingSpaces,L' ');
        const auto text = syntax.Slice(line.source);
        for (std::size_t i = 0; i < text.size(); ++i)
        {
            if (text[i] == L'\r') { result += L'\n'; if (i + 1u < text.size() && text[i + 1u] == L'\n') ++i; }
            else result += text[i] ? text[i] : L'\ufffd';
        }
    }
    if (!result.empty() && result.back() != L'\n') result += L'\n';
    return result;
}
inline std::wstring Info(const MarkdownSyntax& syntax, MarkdownRange range)
{
    std::wstring result;
    for (std::size_t i = 0; i < syntax.Lexemes().size(); ++i)
    {
        const auto& token = syntax.Lexemes()[i];
        const auto first = std::max(range.first,token.source.first), last = std::min(range.last,token.source.last);
        if (first >= last) continue;
        result += first == token.source.first && last == token.source.last ? syntax.LexemeValue(i) : syntax.Slice({first,last});
    }
    return result;
}
inline std::wstring Html(const MarkdownSyntax& syntax)
{
    using Kind = MarkdownBlockKind;
    struct Visit { std::size_t index; bool close; };
    std::vector<Visit> pending{{0u,false}};
    std::wstring result;
    const auto& nodes = syntax.Nodes();
    while (!pending.empty())
    {
        const auto visit = pending.back(); pending.pop_back();
        const auto& node = nodes.at(visit.index);
        const auto kind = node.block.kind;
        const bool tight = kind == Kind::Paragraph && node.parent != MarkdownNoNode && nodes[node.parent].block.kind == Kind::ListItem && nodes[nodes[node.parent].parent].tight;
        const bool ordered = node.listMarker == L'.' || node.listMarker == L')';
        if (visit.close)
        {
            if (kind == Kind::BlockQuote) result += L"</blockquote>\n";
            else if (kind == Kind::List) result += ordered ? L"</ol>\n" : L"</ul>\n";
            else if (kind == Kind::ListItem) result += L"</li>\n";
            else if (kind == Kind::Table)
            { if (node.firstChild != MarkdownNoNode && nodes[node.firstChild].nextSibling != MarkdownNoNode) result += L"</tbody>\n"; result += L"</table>\n"; }
            else if (kind == Kind::TableRow)
            { result += L"</tr>\n"; if (node.tableHeader) { result += L"</thead>\n"; if (node.nextSibling != MarkdownNoNode) result += L"<tbody>\n"; } }
            continue;
        }
        if (node.nextSibling != MarkdownNoNode) pending.push_back({node.nextSibling,false});
        if (kind == Kind::Paragraph || kind == Kind::AtxHeading || kind == Kind::SetextHeading || kind == Kind::TableCell)
        {
            std::wstring tag;
            if (kind == Kind::Paragraph && !tight) tag = L"p";
            else if (kind == Kind::AtxHeading || kind == Kind::SetextHeading) tag = L"h" + std::to_wstring(node.block.heading);
            else if (kind == Kind::TableCell) tag = node.tableHeader ? L"th" : L"td";
            if (!tag.empty())
            {
                result += L"<" + tag;
                if (kind == Kind::TableCell && node.alignment != MarkdownAlignment::None)
                    result += L" align=\"" + std::wstring(node.alignment == MarkdownAlignment::Left ? L"left" : node.alignment == MarkdownAlignment::Center ? L"center" : L"right") + L"\"";
                result += L">";
            }
            if (kind == Kind::Paragraph && node.parent != MarkdownNoNode && nodes[node.parent].firstChild == visit.index && nodes[node.parent].task != MarkdownTask::None)
                result += nodes[node.parent].task == MarkdownTask::Checked ? L"<input checked=\"\" disabled=\"\" type=\"checkbox\"> " : L"<input disabled=\"\" type=\"checkbox\"> ";
            result += Inline(syntax,node.firstInline);
            if (!tag.empty()) result += L"</" + tag + L">\n";
            else if (node.nextSibling != MarkdownNoNode) result += L'\n';
        }
        else if (kind == Kind::FencedCode || kind == Kind::IndentedCode)
        {
            result += L"<pre><code";
            auto info = Info(syntax,node.block.info);
            const auto space = info.find_first_of(L" \t\r\n");
            if (space != std::wstring::npos) info.resize(space);
            if (!info.empty()) result += L" class=\"language-" + Escape(info) + L"\"";
            result += L">" + Escape(Lines(syntax,node)) + L"</code></pre>\n";
        }
        else if (kind == Kind::HtmlBlock) result += Filter(Lines(syntax,node),syntax.Extensions());
        else if (kind == Kind::ThematicBreak) result += L"<hr />\n";
        else if (kind != Kind::ReferenceDefinition)
        {
            if (kind == Kind::BlockQuote) result += L"<blockquote>\n";
            else if (kind == Kind::List)
            { result += ordered ? node.listStart == 1u ? L"<ol>\n" : L"<ol start=\"" + std::to_wstring(node.listStart) + L"\">\n" : L"<ul>\n"; }
            else if (kind == Kind::ListItem)
            {
                result += L"<li>";
                if (node.firstChild != MarkdownNoNode && (!nodes[node.parent].tight || nodes[node.firstChild].block.kind != Kind::Paragraph)) result += L'\n';
            }
            else if (kind == Kind::Table) result += L"<table>\n<thead>\n";
            else if (kind == Kind::TableRow) result += L"<tr>\n";
            pending.push_back({visit.index,true});
            if (node.firstChild != MarkdownNoNode) pending.push_back({node.firstChild,false});
        }
    }
    return result;
}
} // namespace MarkdownHtmlTests
