$enc = [Text.UTF8Encoding]::new($false)
function Patch([string] $path, [string] $old, [string] $new) {
    $text = $enc.GetString([IO.File]::ReadAllBytes($path))
    $crlf = $text.Contains("`r`n")
    if ($crlf) { $old = $old.Replace("`r`n", "`n").Replace("`n", "`r`n"); $new = $new.Replace("`r`n", "`n").Replace("`n", "`r`n") }
    $count = ([regex]::Matches($text, [regex]::Escape($old))).Count
    if ($count -ne 1) { throw "anchor matched $count times in ${path}: $old" }
    [IO.File]::WriteAllText($path, $text.Replace($old, $new), $enc)
}
