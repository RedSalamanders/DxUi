[CmdletBinding()]
param()
Set-StrictMode -Version Latest
$ErrorActionPreference = 'Stop'
$expectedHead = 'cdb3be2cf68d6444de9e66559db44fcfd75c0781'
$taskClone = '/home/a/codex-dpf-s0-s2-portability-20261011'
$taskExternal = '/mnt/c/Users/eric/.codex/tmp/dpf-s0-s2-merge'
$taskCandidate = '/mnt/c/Users/eric/.codex/worktrees/performance-forms/DxUi'
$taskShell = (Get-Process -Id $PID).Path
$PSVersionTable | ConvertTo-Json -Depth 5 | Set-Content (Join-Path $taskExternal 'linux-portability-runtime.receipt.txt')
& git clone --depth 1 --branch codex/performance-forms https://github.com/RedSalamanders/DxUi.git $taskClone
if ($LASTEXITCODE -ne 0) { throw 'Cannot create the isolated Linux source clone.' }
$actualHead = (& git -C $taskClone rev-parse HEAD).Trim()
if ($LASTEXITCODE -ne 0 -or $actualHead -cne $expectedHead) { throw 'Linux baseline is not the exact published candidate.' }
& git -C $taskClone status --porcelain=v1
if ($LASTEXITCODE -ne 0) { throw 'Cannot check the isolated baseline.' }
$results = [Collections.Generic.List[object]]::new()
foreach ($name in @('Test-PerformancePolicy.ps1','Test-PerformanceProfiling.ps1')) {
    $log = Join-Path $taskExternal ('linux-before-' + $name.Replace('.ps1','.log'))
    & $taskShell -NoLogo -NoProfile -File (Join-Path $taskClone "Tools/tests/$name") *> $log
    $exit = $LASTEXITCODE
    $results.Add([ordered]@{ stage='before'; script=$name; source=$actualHead; exitCode=$exit; log=$log })
    Write-Output "Baseline $name exit $exit"
    Get-Content $log -Tail 7
}
foreach ($name in @('Test-PerformancePolicy.ps1','Test-PerformanceProfiling.ps1')) {
    $source = Join-Path $taskCandidate "Tools/tests/$name"
    $destination = Join-Path $taskClone "Tools/tests/$name"
    Copy-Item -LiteralPath $source -Destination $destination
    $sourceHash = (Get-FileHash $source -Algorithm SHA256).Hash
    if ((Get-FileHash $destination -Algorithm SHA256).Hash -cne $sourceHash) { throw 'Linux candidate copy changed source bytes.' }
    $log = Join-Path $taskExternal ('linux-after-' + $name.Replace('.ps1','.log'))
    & $taskShell -NoLogo -NoProfile -File $destination *> $log
    $exit = $LASTEXITCODE
    $results.Add([ordered]@{ stage='after'; script=$name; sourceBase=$actualHead; scriptSha256=$sourceHash; exitCode=$exit; log=$log })
    Write-Output "Candidate $name exit $exit"
    Get-Content $log -Tail 7
    $results | ConvertTo-Json -Depth 7 | Set-Content (Join-Path $taskExternal 'linux-portability-results.receipt.txt')
    if ($exit -ne 0) { throw "Linux candidate fixture failed: $name" }
}
$log = Join-Path $taskExternal 'linux-portability-validate.log'
& $taskShell -NoLogo -NoProfile -File (Join-Path $taskClone 'validate.ps1') *> $log
$exit = $LASTEXITCODE
$results.Add([ordered]@{ stage='candidate-validation'; sourceBase=$actualHead; exitCode=$exit; log=$log })
$results | ConvertTo-Json -Depth 7 | Set-Content (Join-Path $taskExternal 'linux-portability-results.receipt.txt')
Get-Content $log -Tail 25
if ($exit -ne 0) { throw 'Linux candidate full validation failed.' }
Write-Output 'LINUX_PORTABILITY_REVIEW_READY'
