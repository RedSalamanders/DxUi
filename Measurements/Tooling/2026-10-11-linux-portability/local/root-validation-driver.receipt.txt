[CmdletBinding()]
param([Parameter(Mandatory)][string] $Root,[Parameter(Mandatory)][ValidateSet('windows','linux')][string] $HostLabel,[string] $External = $PSScriptRoot)
Set-StrictMode -Version Latest
$ErrorActionPreference = 'Stop'
$taskShell = (Get-Process -Id $PID).Path
$inputPaths = @('.gitattributes','Tools/tests/Test-PerformancePolicy.ps1','Tools/tests/Test-PerformanceProfiling.ps1',
    'Tools/README.md','Specs/Testing/Testing_Validation.md','Specs/Plans/Done/ConsumerUiImprovements_S0_S2_2026-10-04.md',
    'Specs/Plans/WIP/ConsumerUiQualification_2026-10-11.md','Changes/2026-10-11-linux-performance-fixtures.md',
    'Measurements/Tooling/2026-10-11-linux-portability/README.md')
$before = [ordered]@{}
foreach ($path in $inputPaths) { $before[$path] = (Get-FileHash (Join-Path $Root $path) -Algorithm SHA256).Hash }
$taskHead = (& git -C $Root rev-parse HEAD).Trim()
if ($LASTEXITCODE -ne 0 -or $taskHead -cne 'cdb3be2cf68d6444de9e66559db44fcfd75c0781') { throw 'Wrong validation source base.' }
$steps = [Collections.Generic.List[object]]::new()
foreach ($script in @('validate.ps1') + $(if ($HostLabel -eq 'windows') { @('format.ps1') } else { @() })) {
    $log = Join-Path $External "portability-$HostLabel-$($script.Replace('.ps1','.log'))"
    $arguments = @('-NoLogo','-NoProfile','-File',(Join-Path $Root $script))
    if ($script -eq 'format.ps1') { $arguments += '-Check' }
    & $taskShell @arguments *> $log
    $taskExit = $LASTEXITCODE
    $steps.Add([ordered]@{ script=$script; exitCode=$taskExit; log=$log; sha256=(Get-FileHash $log -Algorithm SHA256).Hash })
    Get-Content $log -Tail 15
    if ($taskExit -ne 0) { throw "Portability $HostLabel $script failed." }
}
foreach ($path in $inputPaths) { if ((Get-FileHash (Join-Path $Root $path) -Algorithm SHA256).Hash -cne $before[$path]) { throw "Validation input mutated: $path" } }
[ordered]@{ sourceBase=$taskHead; host=$HostLabel; sourceState='Explicit dirty overlay of listed source/document bytes on exact published base'; inputs=$before; steps=$steps.ToArray(); runtime=$PSVersionTable.PSVersion.ToString(); completedUtc=[DateTime]::UtcNow.ToString('o') } | ConvertTo-Json -Depth 7 | Set-Content (Join-Path $External "portability-$HostLabel-validation.receipt.txt")
Write-Output "PORTABILITY_$($HostLabel.ToUpperInvariant())_VALIDATION_PASS"
