from pathlib import Path
import importlib.util, json, hashlib, difflib, datetime

here = Path(__file__).resolve().parent
draft = here / 'curate-menu-header-ablation-closeout-20260927.py'
spec = importlib.util.spec_from_file_location('draft', draft)
m = importlib.util.module_from_spec(spec)
spec.loader.exec_module(m)

original_validate = m.validate_runs
def validate_runs():
    reports, invocations = original_validate()
    previous_end = None
    for name in m.RUNS:
        run = invocations[name]
        built = m.read_json(m.V2 / name[0] / 'build-receipt.json')
        for a, b in [('executableTextSectionSha256','executableTextSectionSha256'), ('executableSha256Before','executableSha256'), ('librarySha256Before','librarySha256')]:
            assert run[a] == built[b], (name, a)
        start = datetime.datetime.fromisoformat(run['startedUtc'])
        end = datetime.datetime.fromisoformat(run['completedUtc'])
        assert end >= start and (previous_end is None or start >= previous_end), name
        previous_end = end
        unpaired = m.V2 / (name + '.json.comparison.json')
        assert m.sha(unpaired.read_bytes()) == run['unpairedComparisonSha256']
        assert m.read_json(unpaired)['status'] == 'unpaired'
    return reports, invocations
m.validate_runs = validate_runs

def copy_bytes(source, relative, raw_map, *, data=None):
    payload = source.read_bytes() if data is None else data
    leaf = Path(relative).name
    relative = f'raw/reports/{leaf}' if leaf in m.REPORT_NAMES else f'raw/{m.sha(payload).lower()[:24]}.receipt.txt'
    destination = m.OUT / relative
    destination.parent.mkdir(parents=True, exist_ok=True)
    assert not destination.exists() or destination.read_bytes() == payload, relative
    destination.write_bytes(payload)
    raw_map.append({'source':str(source),'path':relative,'sha256':m.sha(payload).lower(),'size':len(payload)})
m.copy_bytes = copy_bytes

def capture_sources(raw_map, library_sets):
    sources = {}
    for variant, repo in [('A',m.A_REPO),('H',m.H_REPO),('B',m.B_REPO)]:
        sources[variant] = {}
        for rel in ['include/DxUi/DxUi.h','src/Controls/DxUi.Menu.cpp']:
            payload = (repo / rel).read_bytes()
            assert m.sha(payload) == library_sets[variant][rel], (variant,rel)
            sources[variant][rel] = payload
            copy_bytes(repo / rel, rel + '.source.txt', raw_map, data=payload)
    for left, right, rel in [('A','H','include/DxUi/DxUi.h'),('H','B','src/Controls/DxUi.Menu.cpp')]:
        patch = ''.join(difflib.unified_diff(sources[left][rel].decode('utf-8').splitlines(keepends=True), sources[right][rel].decode('utf-8').splitlines(keepends=True), fromfile=left+'/'+rel, tofile=right+'/'+rel))
        copy_bytes(Path(f'@derived/{left}-{right}.patch'), 'patch.receipt.txt', raw_map, data=patch.encode('utf-8'))
    copy_bytes(draft, draft.name, raw_map)
    copy_bytes(Path(__file__), Path(__file__).name, raw_map)
m.capture_sources = capture_sources

m.write_packet()
