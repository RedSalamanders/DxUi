function Assert([bool] $Condition, [string] $Message) {
    if (-not $Condition) { throw $Message }
}
function Hash([string] $Path) { (Get-FileHash -LiteralPath $Path -Algorithm SHA256).Hash.ToUpperInvariant() }
function Pe-Text-Hash([string] $Path) {
    $stream = [IO.File]::OpenRead($Path)
    $reader = [IO.BinaryReader]::new($stream)
    try {
        $stream.Position = 0x3c
        $peOffset = $reader.ReadInt32()
        $stream.Position = $peOffset
        Assert ($reader.ReadUInt32() -eq 0x00004550) "Invalid PE signature: $Path"
        $null = $reader.ReadUInt16() # machine
        $sectionCount = $reader.ReadUInt16()
        $stream.Position = $peOffset + 4 + 16
        $optionalHeaderSize = $reader.ReadUInt16()
        $sectionStart = $peOffset + 4 + 20 + $optionalHeaderSize
        $textHash = $null
        for ($i = 0; $i -lt $sectionCount; ++$i) {
            $stream.Position = $sectionStart + ($i * 40)
            $nameBytes = $reader.ReadBytes(8)
            $name = [Text.Encoding]::ASCII.GetString($nameBytes).Trim([char]0)
            $null = $reader.ReadUInt32() # virtual size
            $null = $reader.ReadUInt32() # virtual address
            $rawSize = $reader.ReadUInt32()
            $rawOffset = $reader.ReadUInt32()
            if ($name -eq '.text') {
                Assert ($null -eq $textHash) "Multiple .text sections: $Path"
                $stream.Position = $rawOffset
                $bytes = $reader.ReadBytes([int]$rawSize)
                Assert ($bytes.Length -eq $rawSize) "Truncated .text section: $Path"
                $textHash = [Convert]::ToHexString([Security.Cryptography.SHA256]::HashData($bytes))
            }
        }
        Assert ($null -ne $textHash) "Missing .text section: $Path"
        return $textHash
    } finally { $reader.Dispose(); $stream.Dispose() }
}
function Save-Json([string] $Path, $Value) {
    $Value | ConvertTo-Json -Depth 16 | Set-Content -LiteralPath $Path -Encoding utf8
}
function Read-Json([string] $Path) { Get-Content -Raw -LiteralPath $Path | ConvertFrom-Json -AsHashtable }
function Git-Output([string] $Repo, [string[]] $Arguments) {
    Push-Location -LiteralPath $Repo
    try {
        $result = @(& git @Arguments)
        Assert ($LASTEXITCODE -eq 0) "git $($Arguments -join ' ') failed in $Repo"
        return $result
    } finally { Pop-Location }
}
function Input-Manifest([string] $Repo) {
    @($inputPaths | ForEach-Object {
        $path = Join-Path $Repo $_
        Assert (Test-Path -LiteralPath $path -PathType Leaf) "Missing fixture input: $path"
        [ordered]@{ path = $_; sha256 = Hash $path }
    })
}
function Library-Manifest([string] $Repo) {
    $files = Git-Output $Repo @('ls-files', '--cached', '--others', '--exclude-standard', '--', 'src', 'include', 'Build', 'Directory.Build.props', 'Directory.Build.targets', 'vcpkg.json', 'vcpkg-tool.json')
    @($files | Sort-Object -Unique | ForEach-Object {
        $path = Join-Path $Repo $_
        Assert (Test-Path -LiteralPath $path -PathType Leaf) "Missing library input: $path"
        [ordered]@{ path = $_; sha256 = Hash $path }
    })
}
function Manifest-Fingerprint($Items) {
    $lines = @($Items | ForEach-Object { "$($_.path) $($_.sha256)" })
    $sha = [Security.Cryptography.SHA256]::Create()
    try { [Convert]::ToHexString($sha.ComputeHash([Text.Encoding]::UTF8.GetBytes(($lines -join "`n")))) }
    finally { $sha.Dispose() }
}
function Git-Receipt([string] $Repo, [string] $Folder) {
    $head = @(Git-Output $Repo @('rev-parse', 'HEAD'))[0].Trim()
    $head | Set-Content -LiteralPath (Join-Path $Folder 'head.txt') -Encoding utf8
    $status = Git-Output $Repo @('status', '--porcelain=v1')
    ($status -join "`n") | Set-Content -LiteralPath (Join-Path $Folder 'status.txt') -Encoding utf8
    $sourceDiff = Git-Output $Repo @('diff', '--binary', '--', 'src', 'include', 'Build', 'Directory.Build.props', 'Directory.Build.targets', 'vcpkg.json', 'vcpkg-tool.json')
    ($sourceDiff -join "`n") | Set-Content -LiteralPath (Join-Path $Folder 'library-diff.patch') -Encoding utf8
    return $head, $status
}
function Assert-SameManifest($Left, $Right, [string] $Label) {
    Assert ($Left.Count -eq $Right.Count) "$Label source inventory size differs"
    for ($i = 0; $i -lt $Left.Count; ++$i) {
        Assert ($Left[$i].path -eq $Right[$i].path) "$Label path differs at $i"
        Assert ($Left[$i].sha256 -eq $Right[$i].sha256) "$Label source differs: $($Left[$i].path)"
    }
}
function Changed-Paths($Left, $Right) {
    $a = @{}; foreach ($item in $Left) { $a[$item.path] = $item.sha256 }
    $b = @{}; foreach ($item in $Right) { $b[$item.path] = $item.sha256 }
    @(@($a.Keys) + @($b.Keys) | Sort-Object -Unique | Where-Object {
        (-not $a.ContainsKey($_)) -or (-not $b.ContainsKey($_)) -or ($a[$_] -ne $b[$_])
    })
}
function Assert-OnlyChangedPath($Left, $Right, [string] $Expected, [string] $Label) {
    $changed = @(Changed-Paths $Left $Right)
    Assert (($changed.Count -eq 1) -and ($changed[0] -eq $Expected)) "$Label delta must be only $Expected; found: $($changed -join ', ')"
}
function Assert-ExactHeaderOverlay([string] $AHeader, [string] $HHeader, [string] $BHeader) {
    $a = (Get-Content -Raw -LiteralPath $AHeader) -replace "`r`n", "`n"
    $h = (Get-Content -Raw -LiteralPath $HHeader) -replace "`r`n", "`n"
    $b = (Get-Content -Raw -LiteralPath $BHeader) -replace "`r`n", "`n"
    $pattern = '(?m)^([ \t]*UINT32 secondaryLineCount[ \t]*=[ \t]*0;\n)([ \t]*bool hasBitmapIcon)'
    $matches = [regex]::Matches($a, $pattern)
    Assert ($matches.Count -eq 1) 'Baseline header does not have the expected single overlay anchor'
    $expected = [regex]::Replace($a, $pattern, {
        param($match)
        $match.Groups[1].Value + '    float primaryLayoutWidthDip      = 0.0f;' + "`n" +
            '    float secondaryLayoutWidthDip    = 0.0f;' + "`n" + $match.Groups[2].Value
    })
    Assert ($h -ceq $expected) 'H header is not exactly the two-field candidate-header overlay on A'
    Assert ($b -ceq $h) 'H and B headers are not byte-equivalent after newline normalization'
}
function Compare-Runs([string] $Name, [string] $BeforeName, [string] $AfterName) {
    $before = Join-Path $Evidence "$BeforeName.json"
    $after = Join-Path $Evidence "$AfterName.json"
    $output = Join-Path $Evidence "$Name.json"
    $log = Join-Path $Evidence "$Name.log"
    & $PythonExe (Join-Path $repos.B 'Tools/compare_performance.py') $after --baseline $before --output $output *> $log
    $comparisonExit = $LASTEXITCODE
    $comparisonExit.ToString() | Set-Content -LiteralPath (Join-Path $Evidence "$Name-exit.txt")
    Assert (Test-Path -LiteralPath $output -PathType Leaf) "Missing comparison: $Name"
    $comparison = Read-Json $output
    Assert ($comparison.status -in @('within-noise-budget', 'advice-required')) "Invalid comparison status: $Name/$($comparison.status)"
    Assert (($comparisonExit -eq 0) -eq ($comparison.status -eq 'within-noise-budget')) "Comparison exit/status mismatch: $Name"
    return $comparison
}
