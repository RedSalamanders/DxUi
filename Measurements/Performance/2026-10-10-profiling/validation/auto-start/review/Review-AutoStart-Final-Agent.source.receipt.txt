[CmdletBinding()]
param()
Set-StrictMode -Version Latest
$ErrorActionPreference = 'Stop'
$root = 'C:/Users/eric/.codex/worktrees/performance-forms/DxUi'
$archive = Join-Path $root 'Measurements/Performance/2026-10-10-profiling/validation/auto-start'
$agentPath = Join-Path $PSScriptRoot 'autostart-final-agent-review.receipt.txt'
$agentHash = (Get-FileHash -LiteralPath $agentPath -Algorithm SHA256).Hash
if ($agentHash -cne '7F18FC3548DB7EA217E8679383141A2A1D8C31C25E0AF17C76AD93D21895820A') { throw 'The original agent review changed.' }
$agent = Get-Content -LiteralPath $agentPath -Raw | ConvertFrom-Json -AsHashtable -Depth 64
$inventory = [Collections.Generic.List[string]]::new()
$filesChecked = 0
$rawBindings = 0
foreach ($packet in $agent.externalPackets) {
    $folder = Split-Path -Leaf $packet
    $mirror = if ($folder -ceq 'performance-autostart') { 'performance' } else { $folder.Replace('focus-autostart-','focus-').Replace('-attempt1','') }
    foreach ($file in Get-ChildItem -LiteralPath $packet -File) {
        $hash = (Get-FileHash -LiteralPath $file.FullName -Algorithm SHA256).Hash
        $inventory.Add($folder + '/' + $file.Name + [char]0 + $file.Length.ToString([Globalization.CultureInfo]::InvariantCulture) + [char]0 + $hash)
        $copy = Join-Path $archive ($mirror + '/' + $file.Name)
        if ((Get-Item -LiteralPath $copy).Length -ne $file.Length -or (Get-FileHash -LiteralPath $copy -Algorithm SHA256).Hash -cne $hash) { throw "Repository mirror differs: $folder/$($file.Name)" }
        $filesChecked++
    }
    $manifests = if ($folder -ceq 'performance-autostart') { @('initial-copy.receipt.txt','control-copy.receipt.txt') }
        elseif ($folder -like '*ASanDebug*') { @('copy.receipt.txt','probe-copy.receipt.txt') } else { @('copy.receipt.txt') }
    foreach ($name in $manifests) {
        $bindings = @(Get-Content -LiteralPath (Join-Path $packet $name) -Raw | ConvertFrom-Json -Depth 32)
        foreach ($binding in $bindings) {
            $copy = Join-Path $packet $binding.copy
            if ((Get-Item -LiteralPath $copy).Length -ne $binding.bytes -or (Get-FileHash -LiteralPath $copy -Algorithm SHA256).Hash -cne $binding.sha256) { throw "Immutable raw binding differs: $folder/$($binding.copy)" }
            $rawBindings++
        }
    }
}
$inventory.Sort([StringComparer]::Ordinal)
$inventoryHash = [Convert]::ToHexString([Security.Cryptography.SHA256]::HashData([Text.Encoding]::UTF8.GetBytes($inventory -join [char]10)))
if ($filesChecked -ne 69 -or $rawBindings -ne 53 -or $inventoryHash -cne $agent.retainedInventorySha256) { throw 'The independently reproduced inventory differs from the agent review.' }
$artifacts = @(foreach ($artifact in $agent.actualNamedArtifactHashes) {
    if ((Get-Item -LiteralPath $artifact.path).Length -ne $artifact.bytes -or (Get-FileHash -LiteralPath $artifact.path -Algorithm SHA256).Hash -cne $artifact.sha256) { throw 'A named current native artifact differs.' }
    [ordered]@{ path=$artifact.path; bytes=$artifact.bytes; sha256=$artifact.sha256 }
})
Import-Module (Join-Path $root 'Tools/PerformanceComparison.psm1') -Force
$performance = Join-Path $archive 'performance'
$before = Read-PerformanceReceipt (Join-Path $performance 'before-Release.json')
$after = Read-PerformanceReceipt (Join-Path $performance 'after-Release.json')
$replay = Compare-PerformanceReceipt -Before $before -After $after
$stored = Get-Content -LiteralPath (Join-Path $performance 'after-comparison.receipt.txt') -Raw | ConvertFrom-Json -AsHashtable -Depth 32
if ((ConvertTo-PerformanceComparisonJson $replay ([string][char]10)) -cne (ConvertTo-PerformanceComparisonJson $stored ([string][char]10))) { throw 'The complete frozen comparator replay differs.' }
$flags = @($replay.changes | Where-Object regressed).Count
if ($replay.status -cne 'advice-required' -or $replay.changes.Count -ne 26 -or $flags -ne 10) { throw 'The retained comparison status/counts differ.' }
$medianRows = 0
foreach ($row in $agent.performance.fpsMedianRows) {
    $report = Read-PerformanceReceipt (Join-Path $performance $row.report)
    $scenario = @($report.scenarios | Where-Object name -CEQ $row.scenario)
    if ($scenario.Count -ne 1) { throw 'The retained FPS scenario is ambiguous.' }
    $raw = @($scenario[0].rounds.fps)
    $sorted = @($raw | Sort-Object)
    if ($sorted.Count -ne 5 -or $sorted[2] -ne $row.medianFps -or $sorted[0] -ne $row.minimumFps -or $sorted[4] -ne $row.maximumFps -or ($raw -join '|') -cne ($row.orderedRoundFps -join '|')) { throw 'An agent FPS row differs from raw rounds.' }
    $medianRows++
}
$planPath = Join-Path $root 'Specs/Plans/WIP/ConsumerUiImprovements_2026-10-04.md'
$plan = Get-Content -LiteralPath $planPath -Raw
if (-not $plan.Contains('The remaining native-platform and paired-performance gates') -or $plan.Contains('The remaining focus, native-platform and paired-performance gates')) { throw 'The independently confirmed stale plan sentence remains.' }
$proof = [ordered]@{
    purpose='root-independent-replay-of-final-agent-audit'
    sourceCommit=$agent.sourceCommit
    reviewedUtc=[DateTime]::UtcNow.ToString('o')
    agentReceiptSha256=$agentHash
    immutableFilesAndMirrorsChecked=$filesChecked
    rawCopyBindingsChecked=$rawBindings
    independentlyReproducedInventorySha256=$inventoryHash
    actualNamedArtifactsMatched=$artifacts
    pureComparatorCells=$replay.changes.Count
    pureComparatorTimingFlags=$flags
    originalComparisonStatus=$replay.status
    rawFpsRowsReproduced=$medianRows
    finding=[ordered]@{ original=$agent.findings[0]; resolution='The lead verified the completed local x64 focus checkbox and removed the stale focus item from the closing remaining-gates sentence.'; correctedPlanSha256=(Get-FileHash -LiteralPath $planPath -Algorithm SHA256).Hash }
    scope='Immutable file/copy/inventory replay, eight named current native artifact byte checks, complete pure frozen comparator replay and eight raw FPS rows. Previous independently retained root reviews own the full native scope and actual installed tool/SDK/dependency state guards. No native rerun or installed SDK scan.'
    acceptance='Performance flags and separate native-platform, physical-input, consumer and required-CI gates remain open.'
    reviewerScriptSha256=(Get-FileHash -LiteralPath $PSCommandPath -Algorithm SHA256).Hash
}
[IO.File]::WriteAllText((Join-Path $PSScriptRoot 'autostart-final-agent-root-replay.receipt.txt'),((ConvertTo-Json -InputObject $proof -Depth 32)+[char]10),[Text.UTF8Encoding]::new($false))
Write-Output "Root replay passed: $filesChecked immutable files/mirrors, $rawBindings raw bindings, $($artifacts.Count) current artifacts, 26 comparator cells/10 flags, $medianRows raw FPS rows. Stale plan sentence corrected; performance remains open."
