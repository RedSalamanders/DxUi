[CmdletBinding()]
param()
Set-StrictMode -Version Latest
$ErrorActionPreference = 'Stop'
function Get-Digest([string]$Value) {
    [Convert]::ToHexString([Security.Cryptography.SHA256]::HashData([Text.Encoding]::UTF8.GetBytes($Value))).ToLowerInvariant()
}
$base = Join-Path $PSScriptRoot 'build-autostart-final'
$bindings = 0
$envelopes = 0
$buildIdentities = 0
foreach ($platform in 'x64','ARM64') {
    foreach ($configuration in 'Debug','Release','ASan Debug') {
        $folder = Join-Path $base ($platform+'-'+$configuration.Replace(' ',''))
        $review = Get-Content -LiteralPath (Join-Path $folder 'root-review.receipt.txt') -Raw | ConvertFrom-Json -AsHashtable -Depth 32
        $state = $review.actualBuildState
        if ($review.platform -cne $platform -or $review.configuration -cne $configuration -or $review.exitCode -ne 0 -or
            $review.nativeRuntimeQualified -or $state.sourceDirty -or $state.sourceCommit -cne '02961a88321c75ccd39090873f48977206d6dfdb' -or
            $state.Count -ne 26 -or $state.buildIdentityVersion -ne 2 -or $state.identityStatus -cne 'verifiable' -or $state.identityScope -cne 'full') { throw 'Wrong frozen build snapshot.' }
        foreach ($copy in $review.copies) {
            $file = Join-Path $folder $copy.copy
            if ((Get-Item -LiteralPath $file).Length -ne $copy.bytes -or (Get-FileHash -LiteralPath $file -Algorithm SHA256).Hash -cne $copy.sha256) { throw 'Build copy binding differs.' }
            $bindings++
        }
        $identity = Get-Digest (@('dxui-performance-build-v2',$state.buildIdentityVersion,$state.resolvedBuildIdentity,$state.buildInputIdentity,$state.toolchainIdentity,$state.buildArtifactIdentity) -join "`n")
        if ($identity -cne $state.buildAttestationIdentity) { throw 'The stored complete v2 build identity does not replay.' }
        foreach ($name in 'build.receipt.txt','incremental.receipt.txt') {
            $envelope = Get-Content -LiteralPath (Join-Path $folder $name) -Raw | ConvertFrom-Json -AsHashtable
            $completed = if ($envelope.completedUtc -is [DateTime]) { $envelope.completedUtc.ToUniversalTime().ToString('o') } else { [string]$envelope.completedUtc }
            $digest = Get-Digest ($envelope.identity+"`n"+$completed+"`nPASSED")
            if ($envelope.version -ne 1 -or $envelope.outcome -cne 'PASSED' -or $envelope.digest -cne $digest) { throw 'A stored build or incremental envelope digest does not replay.' }
            if ($name -ceq 'build.receipt.txt') {
                if ($envelope.identity -cne $identity) { throw 'The build envelope does not bind the complete stored v2 state.' }
                $buildIdentities++
            }
            $envelopes++
        }
    }
}
$receipt = [ordered]@{ purpose='root-independent-archived-six-build-replay'; sourceCommit='02961a88321c75ccd39090873f48977206d6dfdb';
    configurations=6; rawBindings=$bindings; envelopeDigests=$envelopes; fullBuildIdentities=$buildIdentities; archivedIdentitiesMatched=$true; nativeRuntimeQualified=$false;
    scope='Archived consistency only; original snapshot root reviews record actual live build input, tool, SDK, dependency and artifact checks. Incremental intermediate trees are not archived or reconstructed; only their original envelope digest and copy binding are replayed.';
    reviewerScriptSha256=(Get-FileHash -LiteralPath $PSCommandPath -Algorithm SHA256).Hash; reviewedUtc=[DateTime]::UtcNow.ToString('o') }
[IO.File]::WriteAllText((Join-Path $PSScriptRoot 'autostart-build-independent-review.receipt.txt'), ((ConvertTo-Json -InputObject $receipt -Depth 12)+[char]10), [Text.UTF8Encoding]::new($false))
Write-Output "PASS root build replay: six complete v2 identities, $bindings raw bindings, $envelopes stored envelope digests; ARM64 build scope only."
