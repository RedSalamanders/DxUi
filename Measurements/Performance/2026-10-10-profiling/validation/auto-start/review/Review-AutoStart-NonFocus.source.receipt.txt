[CmdletBinding()]
param()
Set-StrictMode -Version Latest
$ErrorActionPreference = 'Stop'
$base = $PSScriptRoot
$suites = @('Foundation','Embedded','Grid','Theme','Control','MenuExitLifetime','NewControls','EditorControls','TextField','MultilineText','ReadOnly','ComboBox','Tree','Tooltip','Rendering','Animation','Accessibility','WindowHost','InteractiveLease')
$newCases = @('TestApprovedAutomaticLeaseRunsAndRecordsItsCountdown','TestAutomaticAnswerNeedsAnApprovedRequestAndPositiveCountdown','TestApprovedAutomaticLeaseCanBeCancelledAndInterrupted','TestApprovedAutomaticLeaseRefusesADesktopLostDuringTheCountdown','TestApprovedAutomaticLeaseRestoresAfterWarningAndChildFailures')
$copyCount = 0
$envelopeCount = 0
$profiles = @(foreach ($configuration in 'Debug','Release','ASan Debug') {
    $folder = Join-Path $base ('nonfocus-autostart-'+$configuration.Replace(' ',''))
    $copies = @(Get-Content -LiteralPath (Join-Path $folder 'copy.receipt.txt') -Raw | ConvertFrom-Json -Depth 16)
    foreach ($entry in $copies) {
        $copy = Join-Path $folder $entry.copy
        if ((Get-Item -LiteralPath $copy).Length -ne $entry.bytes -or (Get-FileHash -LiteralPath $copy -Algorithm SHA256).Hash -cne $entry.sha256) { throw "Archived raw bytes differ: $configuration/$($entry.copy)" }
        $copyCount++
    }
    $state = Get-Content -LiteralPath (Join-Path $folder 'post-native.state.receipt.txt') -Raw | ConvertFrom-Json -AsHashtable -Depth 32
    $performance = Get-Content -LiteralPath (Join-Path $folder 'canonical-performance.json') -Raw | ConvertFrom-Json -AsHashtable -Depth 64
    if ($state.Count -ne 26 -or $state.sourceDirty -or $state.sourceCommit -cne '02961a88321c75ccd39090873f48977206d6dfdb') { throw 'Wrong captured source/profile state.' }
    foreach ($key in $state.Keys) {
        $stateValue = ConvertTo-Json -InputObject $state[$key] -Depth 32 -Compress
        $reportValue = ConvertTo-Json -InputObject $performance[$key] -Depth 32 -Compress
        if ($stateValue -cne $reportValue) { throw "Post-state differs from the native report: $configuration/$key" }
    }
    $runner = @(Get-Content -LiteralPath (Join-Path $folder 'runner.log'))
    $running = @($runner | Where-Object { $_ -cmatch '^Running (\w+) \(x64 ' } | ForEach-Object { [regex]::Match($_,'^Running (\w+)').Groups[1].Value })
    $passed = @($runner | Where-Object { $_ -cmatch '^PASS (\w+) \(0 capability skips recorded\)$' } | ForEach-Object { [regex]::Match($_,'^PASS (\w+)').Groups[1].Value })
    if (($running -join '|') -cne ($suites -join '|') -or ($passed -join '|') -cne ($suites -join '|')) { throw 'The runner did not execute the ordered full native inventory.' }
    foreach ($suite in $suites) {
        $receipt = Get-Content -LiteralPath (Join-Path $folder "$suite.receipt.txt") -Raw | ConvertFrom-Json -AsHashtable -Depth 32
        $scoped = Get-Content -LiteralPath (Join-Path $folder "$suite.scoped.receipt.txt") -Raw | ConvertFrom-Json -AsHashtable
        $rootReview = Get-Content -LiteralPath (Join-Path $folder 'root-review.receipt.txt') -Raw | ConvertFrom-Json -AsHashtable -Depth 32
        $recorded = @($rootReview.suites | Where-Object suite -CEQ $suite)
        if ($receipt.suite -cne $suite -or $receipt.configuration -cne $configuration -or $receipt.exitCode -ne 0 -or
            @($receipt.skips).Count -or $receipt.Contains('tests') -or $receipt.Contains('interactive') -or
            $recorded.Count -ne 1 -or $recorded[0].executableSha256 -cne $receipt.sha256 -or $recorded[0].scopedIdentity -cne $scoped.identity) { throw 'A full-suite receipt differs from the retained root review.' }
        $lines = @(Get-Content -LiteralPath (Join-Path $folder "$suite.log"))
        $positive = switch ($suite) {
            Foundation {'^Foundation tests: PASS$'}
            Embedded {'^PASS [0-9]+ checks; 1000 warm composites [0-9]+ us; C\+\+ allocations [0-9]+; changed toggle pixels [0-9]+$'}
            MenuExitLifetime {'^\[EXIT\] MenuExitLifetime: active menu is dismissed before the process-exit host sweep$'}
            default {'^All DxUi tests passed\.$'}
        }
        if (@($lines | Where-Object { $_ -cmatch $positive }).Count -ne 1 -or @($lines | Where-Object { $_ -match '^SKIPPED:|^TIMEOUT:|^FAILED:' }).Count) { throw 'Native suite completion markers are not valid.' }
    }
    $leaseLines = @(Get-Content -LiteralPath (Join-Path $folder 'InteractiveLease.log'))
    foreach ($case in $newCases) {
        if (@($leaseLines | Where-Object { $_ -ceq "  [START] $case" }).Count -ne 1 -or
            @($leaseLines | Where-Object { $_ -cmatch ('^  \[DONE\] '+[regex]::Escape($case)+' \(') }).Count -ne 1) { throw 'A new lease case did not start and complete.' }
    }
    foreach ($name in @($suites | ForEach-Object { "$_.scoped.receipt.txt" }) + @('scoped-build.receipt.txt','performance-build.receipt.txt','Tooling.scoped.receipt.txt')) {
        $envelope = Get-Content -LiteralPath (Join-Path $folder $name) -Raw | ConvertFrom-Json -AsHashtable
        $completed = if ($envelope.completedUtc -is [DateTime]) { $envelope.completedUtc.ToUniversalTime().ToString('o') } else { [string]$envelope.completedUtc }
        $digest = [Convert]::ToHexString([Security.Cryptography.SHA256]::HashData([Text.Encoding]::UTF8.GetBytes($envelope.identity+"`n"+$completed+"`nPASSED"))).ToLowerInvariant()
        if ($envelope.version -ne 1 -or $envelope.outcome -cne 'PASSED' -or $envelope.digest -cne $digest) { throw 'A stored envelope digest does not replay.' }
        $envelopeCount++
    }
    if ($configuration -ceq 'ASan Debug') {
        $probe = Get-Content -LiteralPath (Join-Path $folder 'AddressSanitizer.receipt.txt') -Raw | ConvertFrom-Json -AsHashtable
        if ($probe.exitCode -ne 1 -or -not $probe.detected -or
            @(Select-String -LiteralPath (Join-Path $folder 'AddressSanitizer.log') -Pattern '^==.*ERROR: AddressSanitizer: heap-use-after-free').Count -ne 1) { throw 'The isolated expected sanitizer probe is not diagnosed.' }
    }
    [ordered]@{ configuration=$configuration; suites=$suites.Count; skips=0; newCases=$newCases.Count; copiedOriginals=$copies.Count; capturedStateFields=$state.Count }
})
if ($copyCount -ne 191 -or $envelopeCount -ne 66) { throw 'The archived audit inventory changed.' }
$review = [ordered]@{ purpose='root-independent-replay-of-agent-archived-noninteractive-audit'; sourceCommit='02961a88321c75ccd39090873f48977206d6dfdb';
    copiedOriginals=$copyCount; envelopeDigests=$envelopeCount; profiles=$profiles; verified=true;
    scope='Archived consistency and positive complete suite execution; actual live provenance was checked by each original root snapshot helper. This replay does not execute native work or establish focus/performance acceptance.';
    reviewerScriptSha256=(Get-FileHash -LiteralPath $PSCommandPath -Algorithm SHA256).Hash; reviewedUtc=[DateTime]::UtcNow.ToString('o') }
[IO.File]::WriteAllText((Join-Path $base 'autostart-nonfocus-independent-review.receipt.txt'), ((ConvertTo-Json -InputObject $review -Depth 12)+[char]10), [Text.UTF8Encoding]::new($false))
Write-Output "PASS root replay: $copyCount raw copy bindings, $envelopeCount stored envelope digests, 57 full suite executions and all five new cases per profile."
