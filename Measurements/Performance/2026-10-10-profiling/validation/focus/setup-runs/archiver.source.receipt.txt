[CmdletBinding()]
param(
    [Parameter(Mandatory)][string] $SnapshotDirectory,
    [Parameter(Mandatory)][ValidatePattern('^[A-Za-z0-9-]+$')][string] $DestinationName,
    [Parameter(Mandatory)][string] $ReadmeFile
)
Set-StrictMode -Version Latest
$ErrorActionPreference = 'Stop'
$root = 'C:/Users/eric/.codex/worktrees/performance-forms/DxUi'
$privateRoot = [IO.Path]::GetFullPath($PSScriptRoot).TrimEnd('\','/')+[IO.Path]::DirectorySeparatorChar
$source = [IO.Path]::GetFullPath($SnapshotDirectory)
if (-not $source.StartsWith($privateRoot,[StringComparison]::OrdinalIgnoreCase) -or -not (Test-Path -LiteralPath $source -PathType Container)) {throw 'Snapshot is outside the intended private evidence directory or missing.'}
$archiveRoot = [IO.Path]::GetFullPath((Join-Path $root 'Measurements/Performance/2026-10-10-profiling/validation/focus/setup-runs'))
$destination = [IO.Path]::GetFullPath((Join-Path $archiveRoot $DestinationName))
if (-not $destination.StartsWith(($archiveRoot+[IO.Path]::DirectorySeparatorChar),[StringComparison]::OrdinalIgnoreCase)) {throw 'Archive target escaped its intended directory.'}
if (Test-Path -LiteralPath $destination) {throw 'Archive target already exists; preserve its exact bytes.'}
if (-not (Test-Path -LiteralPath (Join-Path $source 'root-review.receipt.txt') -PathType Leaf)) {throw 'Snapshot has no completed root review.'}
if (@(Get-ChildItem -LiteralPath $source -Directory -Force).Count) {throw 'Expected a flat native snapshot.'}
if (-not (Test-Path -LiteralPath $ReadmeFile -PathType Leaf)) {throw 'A scenario and source explanation is required.'}
$files = @(Get-ChildItem -LiteralPath $source -File -Force | Sort-Object Name)
if (@($files | Where-Object Name -in @('README.md','archive-copy.receipt.txt')).Count) {throw 'Snapshot contains a reserved archive metadata name.'}
New-Item -ItemType Directory -Path $destination -Force | Out-Null
$copies = @(foreach ($file in $files) {
    $hash = (Get-FileHash -LiteralPath $file.FullName -Algorithm SHA256).Hash
    $length = $file.Length
    $copy = Join-Path $destination $file.Name
    Copy-Item -LiteralPath $file.FullName -Destination $copy
    if ((Get-Item -LiteralPath $copy).Length -ne $length -or (Get-FileHash -LiteralPath $copy -Algorithm SHA256).Hash -cne $hash -or
        (Get-FileHash -LiteralPath $file.FullName -Algorithm SHA256).Hash -cne $hash) {throw "Archive copy differs: $($file.Name)"}
    [ordered]@{original=$file.FullName;copy=$file.Name;bytes=$length;sha256=$hash}
})
[IO.File]::WriteAllText((Join-Path $destination 'archive-copy.receipt.txt'),((ConvertTo-Json -InputObject $copies -Depth 8)+[char]10),[Text.UTF8Encoding]::new($false))
Copy-Item -LiteralPath $ReadmeFile -Destination (Join-Path $destination 'README.md')
if ((Get-FileHash -LiteralPath $ReadmeFile -Algorithm SHA256).Hash -cne (Get-FileHash -LiteralPath (Join-Path $destination 'README.md') -Algorithm SHA256).Hash) {throw 'Scenario explanation copy differs.'}
Write-Output "Archived $($copies.Count) exact bindings: $destination"
