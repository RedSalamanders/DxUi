[CmdletBinding()]
param(
    [Parameter(Mandatory)][ValidateSet('Debug','Release','ASan Debug')][string] $Configuration,
    [Parameter(Mandatory)][int] $ParentExitCode,
    [switch] $ReviewExisting
)
Set-StrictMode -Version Latest
$ErrorActionPreference = 'Stop'
$root = 'C:/Users/eric/.codex/worktrees/performance-forms/DxUi'
$tag = $Configuration.Replace(' ', '')
$output = Join-Path $PSScriptRoot "nonfocus-setup-$tag"
if ($ReviewExisting) {
    if (-not (Test-Path -LiteralPath $output -PathType Container)) {throw "Snapshot does not exist: $output"}
    if (Test-Path -LiteralPath (Join-Path $output 'root-review.receipt.txt')) {throw 'A successful review already exists; preserve it.'}
} elseif (Test-Path -LiteralPath $output) { throw "Snapshot output already exists: $output" }
$suites = @('Foundation','Embedded','Grid','Theme','Control','MenuExitLifetime','NewControls','EditorControls','TextField','MultilineText','ReadOnly','ComboBox','Tree','Tooltip','Rendering','Animation','Accessibility','WindowHost','InteractiveLease')
$reports = Join-Path $root '.build/reports'
$logs = Join-Path $root '.build/logs'
$runnerLog = Join-Path $logs "profiling-focus-setup-noninteractive-$tag.txt"
if ($ParentExitCode -ne 0) { throw "Affected run failed with exit $ParentExitCode; preserve its raw output and investigate before snapshot acceptance." }
$first = Get-Content -LiteralPath (Join-Path $reports "Foundation-x64-$Configuration.json") -Raw | ConvertFrom-Json -AsHashtable -Depth 32
$performancePath = [string]$first.performanceReport
$sources = [Collections.Generic.List[object]]::new()
$sources.Add([pscustomobject]@{Original=$runnerLog; Copy='runner.log'})
foreach ($suite in $suites) {
    $sources.Add([pscustomobject]@{Original=(Join-Path $reports "$suite-x64-$Configuration.json"); Copy="$suite.receipt.txt"})
    $sources.Add([pscustomobject]@{Original=(Join-Path $logs "test-$suite-x64-$Configuration.log"); Copy="$suite.log"})
    $sources.Add([pscustomobject]@{Original=(Join-Path $reports "scoped-tests/x64-$tag/$suite.json"); Copy="$suite.scoped.receipt.txt"})
}
$sources.Add([pscustomobject]@{Original=$performancePath; Copy='canonical-performance.json'})
$sources.Add([pscustomobject]@{Original=($performancePath+'.comparison.json'); Copy='canonical-performance.comparison.receipt.txt'})
$sources.Add([pscustomobject]@{Original=(Join-Path $reports "scoped-tests/x64-$tag/build.json"); Copy='scoped-build.receipt.txt'})
$sources.Add([pscustomobject]@{Original=(Join-Path $reports "performance-build/x64-$tag/build.json"); Copy='performance-build.receipt.txt'})
$sources.Add([pscustomobject]@{Original=(Join-Path $reports 'scoped-tests/independent/Tooling.json'); Copy='Tooling.scoped.receipt.txt'})
if ($Configuration -ceq 'ASan Debug') {
    $sources.Add([pscustomobject]@{Original=(Join-Path $reports 'AddressSanitizer-x64.json'); Copy='AddressSanitizer.receipt.txt'})
    $sources.Add([pscustomobject]@{Original=(Join-Path $logs 'test-AddressSanitizer-x64.log'); Copy='AddressSanitizer.log'})
}
if ($ReviewExisting) {
    $copies = @(Get-Content -LiteralPath (Join-Path $output 'copy.receipt.txt') -Raw | ConvertFrom-Json -AsHashtable)
    if ($copies.Count -ne $sources.Count) {throw 'Retained raw inventory differs from the declared inventory.'}
    for ($index=0; $index -lt $copies.Count; $index++) {
        $row = $copies[$index]
        if ($row.original -cne [IO.Path]::GetFullPath($sources[$index].Original) -or $row.copy -cne $sources[$index].Copy) {throw 'Raw inventory paths differ.'}
        foreach ($path in @($row.original,(Join-Path $output $row.copy))) {
            if ((Get-Item -LiteralPath $path).Length -ne $row.bytes -or (Get-FileHash -LiteralPath $path -Algorithm SHA256).Hash -cne $row.sha256) {throw "Retained raw bytes changed: $path"}
        }
    }
} else {
New-Item -ItemType Directory -Path $output | Out-Null
$copies = @(foreach ($source in $sources) {
    $original = [IO.Path]::GetFullPath($source.Original)
    $bytes = (Get-Item -LiteralPath $original).Length
    $hash = (Get-FileHash -LiteralPath $original -Algorithm SHA256).Hash
    $copy = Join-Path $output $source.Copy
    Copy-Item -LiteralPath $original -Destination $copy
    if ((Get-Item -LiteralPath $copy).Length -ne $bytes -or
        (Get-FileHash -LiteralPath $copy -Algorithm SHA256).Hash -cne $hash -or
        (Get-FileHash -LiteralPath $original -Algorithm SHA256).Hash -cne $hash) { throw "Snapshot changed bytes: $original" }
    [ordered]@{original=$original; copy=$source.Copy; bytes=$bytes; sha256=$hash}
})
[IO.File]::WriteAllText((Join-Path $output 'copy.receipt.txt'), ((ConvertTo-Json -InputObject $copies -Depth 8)+[char]10), [Text.UTF8Encoding]::new($false))
}
Import-Module (Join-Path $root 'Tools/PerformancePolicy.psm1') -Force
$state = Get-PerformanceMeasurementState -Root $root -Platform x64 -Configuration $Configuration
$performance = Get-Content -LiteralPath (Join-Path $output 'canonical-performance.json') -Raw | ConvertFrom-Json -AsHashtable -Depth 32
$before = [ordered]@{}
foreach ($key in $state.Keys) {
    if (-not $performance.Contains($key)) { throw "Missing parent benchmark provenance: $key" }
    $before[$key] = $performance[$key]
}
Assert-PerformanceMeasurementState -Before $before -After $state
Assert-PerformanceBuildAttestation -Root $root -Platform x64 -Configuration $Configuration -State $state
[IO.File]::WriteAllText((Join-Path $output 'post-native.state.receipt.txt'), ((ConvertTo-Json -InputObject $state -Depth 12)+[char]10), [Text.UTF8Encoding]::new($false))
$comparison = Get-Content -LiteralPath (Join-Path $output 'canonical-performance.comparison.receipt.txt') -Raw | ConvertFrom-Json -AsHashtable
if ($comparison.status -cne 'unpaired') { throw 'Parent performance observations are expected to be unpaired.' }
Import-Module (Join-Path $root 'Tools/ScopedTesting.psm1') -Force
if (@(Get-ScopedInstrumentation).Count) { throw 'Instrumented results cannot establish ordinary scope acceptance.' }
$commonNative = Get-ScopedRunIdentity $root x64 $Configuration '(common)'
$buildIdentity = Get-ScopedDigest ((Get-ScopedBuildInputIdentity $root x64)+"`n"+(Get-ScopedArtifactIdentity $root x64 $Configuration))
if (-not (Test-ScopedReceipt (Join-Path $output 'scoped-build.receipt.txt') $buildIdentity)) { throw 'Scoped build receipt does not bind current inputs and artifacts.' }
$commonTooling = Get-ScopedDigest ((Get-ScopedSourceIdentity $root)+"`n"+(Get-ScopedEnvironmentIdentity)+"`n"+(Get-ScopedToolIdentity))
if (-not (Test-ScopedReceipt (Join-Path $output 'Tooling.scoped.receipt.txt') (Get-ScopedDigest ($commonTooling+"`nTooling")))) { throw 'Tooling receipt is not current and digest-valid.' }
$checks = @(foreach ($suite in $suites) {
    $receipt = Get-Content -LiteralPath (Join-Path $output "$suite.receipt.txt") -Raw | ConvertFrom-Json -AsHashtable -Depth 32
    $scoped = Get-Content -LiteralPath (Join-Path $output "$suite.scoped.receipt.txt") -Raw | ConvertFrom-Json -AsHashtable
    $expectedName = switch ($suite) { Foundation {'DxUi.FoundationTests.exe'} Embedded {'DxUi.EmbeddedTests.exe'} default {'DxUi.ControlTests.exe'} }
    $expected = [IO.Path]::GetFullPath((Join-Path $root ".build/x64/$Configuration/$expectedName"))
    $lines = @(Get-Content -LiteralPath (Join-Path $output "$suite.log"))
    $completePattern = switch ($suite) {
        Foundation {'^Foundation tests: PASS$'}
        Embedded {'^PASS [0-9]+ checks; 1000 warm composites [0-9]+ us; C\+\+ allocations [0-9]+; changed toggle pixels [0-9]+$'}
        MenuExitLifetime {'^\[EXIT\] MenuExitLifetime: active menu is dismissed before the process-exit host sweep$'}
        default {'^All DxUi tests passed\.$'}
    }
    $scopeIdentity = Get-ScopedDigest ($commonNative+"`n"+$suite)
    if ($receipt.suite -cne $suite -or $receipt.configuration -cne $Configuration -or $receipt.platform -cne 'x64' -or
        $receipt.nativeArchitecture -cne 'X64' -or $receipt.exitCode -ne 0 -or @($receipt.skips).Count -ne 0 -or
        $receipt.Contains('tests') -or ($receipt.Contains('interactive') -and $receipt.interactive) -or
        [IO.Path]::GetFullPath($receipt.executable) -cne $expected -or
        $receipt.sha256 -cne (Get-FileHash -LiteralPath $expected -Algorithm SHA256).Hash -or
        $receipt.performanceReport -cne $performancePath -or $receipt.performanceExecutableSha256 -cne $state.executableSha256 -or
        $receipt.performanceComparison -cne 'unpaired' -or
        -not (Test-ScopedReceipt (Join-Path $output "$suite.scoped.receipt.txt") $scopeIdentity) -or
        @($lines | Where-Object {$_ -match $completePattern}).Count -ne 1 -or
        @($lines | Where-Object {$_ -match '^SKIPPED:|^TIMEOUT:|^FAILED:'}).Count -ne 0) { throw "Invalid complete noninteractive receipt/log: $suite" }
    [ordered]@{suite=$suite; exitCode=0; skips=0; executableSha256=$receipt.sha256; scopedIdentity=$scoped.identity}
})
$asanDetected = $null
if ($Configuration -ceq 'ASan Debug') {
    $probe = Get-Content -LiteralPath (Join-Path $output 'AddressSanitizer.receipt.txt') -Raw | ConvertFrom-Json -AsHashtable
    $probeExe = [IO.Path]::GetFullPath((Join-Path $root ".build/x64/$Configuration/DxUi.FoundationTests.exe"))
    if ($probe.suite -cne 'AddressSanitizer' -or $probe.configuration -cne $Configuration -or $probe.platform -cne 'x64' -or
        $probe.nativeArchitecture -cne 'X64' -or [IO.Path]::GetFullPath($probe.executable) -cne $probeExe -or
        $probe.sha256 -cne (Get-FileHash -LiteralPath $probeExe -Algorithm SHA256).Hash -or $probe.exitCode -eq 0 -or
        -not $probe.detected -or -not (Select-String -LiteralPath (Join-Path $output 'AddressSanitizer.log') -SimpleMatch 'AddressSanitizer: heap-use-after-free')) {
        throw 'ASan probe companions do not establish diagnosis of the deliberate isolated use-after-free.'
    }
    $asanDetected = $true
}
$review = [ordered]@{purpose='root-verified-affected-noninteractive-diagnostic-binaries'; configuration=$Configuration; platform='x64'; parentExitCode=$ParentExitCode;
    sourceCommit=$state.sourceCommit; sourceDirty=$state.sourceDirty; sourceFingerprint=$state.sourceFingerprint; exactPostNativeIdentityMatched=$true;
    suites=$checks; copiedOriginals=$copies.Count; reviewedExistingRawCopies=[bool]$ReviewExisting; scopedBuildAndToolingDigestsValid=$true; deliberateAsanProbeDetected=$asanDetected; performanceAcceptance='unpaired-observation-only';
    reviewerScriptSha256=(Get-FileHash -LiteralPath $PSCommandPath -Algorithm SHA256).Hash; reviewedUtc=[DateTime]::UtcNow.ToString('o')}
[IO.File]::WriteAllText((Join-Path $output 'root-review.receipt.txt'), ((ConvertTo-Json -InputObject $review -Depth 12)+[char]10), [Text.UTF8Encoding]::new($false))
Write-Output "PASS ${Configuration}: nineteen complete noninteractive suites, zero skips, exact executable and full post-native provenance; $($copies.Count) original/copy bindings."
