Set-StrictMode -Version Latest
$ErrorActionPreference='Stop'
$root='C:/Users/eric/.codex/worktrees/performance-forms/DxUi'
$directory=Join-Path $root 'Measurements/Performance/2026-10-10-profiling/validation/focus/setup-diagnostics/builds-final'
$reviewPath=Join-Path $directory 'root-review.receipt.txt'
$initialPath=Join-Path $directory 'initial-review.receipt.txt'
if (Test-Path -LiteralPath $initialPath) {throw 'Initial matrix review is already preserved.'}
$review=Get-Content -LiteralPath $reviewPath -Raw | ConvertFrom-Json -AsHashtable -Depth 32
$initialHash=(Get-FileHash -LiteralPath $reviewPath -Algorithm SHA256).Hash
Copy-Item -LiteralPath $reviewPath -Destination $initialPath
if ((Get-FileHash -LiteralPath $initialPath -Algorithm SHA256).Hash -cne $initialHash) {throw 'Initial matrix review copy differs.'}
$profiles=@($review.profiles | Where-Object {$_ -is [Collections.IDictionary]})
if ($profiles.Count -ne 6 -or @($profiles | ForEach-Object {"$($_.platform)/$($_.configuration)"} | Sort-Object -Unique).Count -ne 6) {throw 'Expected exactly six distinct build descriptors.'}
$rows=@(Get-Content -LiteralPath (Join-Path $directory 'copy.receipt.txt') -Raw | ConvertFrom-Json -AsHashtable)
foreach ($row in $rows) {
    $path=Join-Path $directory $row.copy
    if ((Get-Item -LiteralPath $path).Length -ne $row.bytes -or (Get-FileHash -LiteralPath $path -Algorithm SHA256).Hash -cne $row.sha256) {throw "Retained matrix bytes differ: $($row.copy)"}
}
Import-Module (Join-Path $root 'Tools/PerformancePolicy.psm1') -Force
foreach ($profile in $profiles) {
    $tag=$profile.configuration.Replace(' ','')
    $state=Get-Content -LiteralPath (Join-Path $directory "$($profile.platform)-$tag.state.receipt.txt") -Raw | ConvertFrom-Json -AsHashtable -Depth 32
    foreach ($key in @('platform','configuration','buildIdentityVersion','buildAttestationIdentity','buildInputIdentity','buildArtifactIdentity','nativeArchitecture')) {
        if ([string]$profile[$key] -cne [string]$state[$key]) {throw "Matrix descriptor/state mismatch: $($profile.platform)/$tag/$key"}
    }
    $identity=Get-PerformanceBuildAttestationIdentity -State $state -ArtifactIdentity $state.buildArtifactIdentity
    Import-Module (Join-Path $root 'Tools/ScopedTesting.psm1') -Force
    if (-not (Test-ScopedReceipt (Join-Path $directory "$($profile.platform)-$tag.build.receipt.txt") $identity)) {throw 'Captured matrix build receipt is not digest-valid against its captured state.'}
    if ($profile.platform -ceq 'x64') {
        $nativeSource=Join-Path $root "Measurements/Performance/2026-10-10-profiling/validation/focus/setup-runs/noninteractive-$tag/post-native.state.receipt.txt"
        if ((Get-FileHash -LiteralPath $nativeSource).Hash -cne (Get-FileHash -LiteralPath (Join-Path $directory "x64-$tag.state.receipt.txt")).Hash) {throw 'Captured x64 state differs from the actual post-native state.'}
    } elseif ($profile.runtimeQualification -cne 'cross-build only; native execution not performed') {throw 'ARM64 runtime qualification is overstated.'}
}
$fixed=[ordered]@{purpose='corrected-source-six-builds-and-captured-v2-attestations';profiles=$profiles;originalCopyBindings=$rows.Count;
    initialReviewSha256=$initialHash;correction='Removed six progress strings mistakenly captured beside the six build descriptors; original metadata retained. No build/runtime result changed.';
    capturedBuildReceiptDigestsValid=$true;runtimeScope='x64 noninteractive evidence stored separately; no ARM64 or foreground execution by this checker';reviewedUtc=[DateTime]::UtcNow.ToString('o')}
[IO.File]::WriteAllText($reviewPath,((ConvertTo-Json -InputObject $fixed -Depth 16)+[char]10),[Text.UTF8Encoding]::new($false))
$sourceCopy=Join-Path $directory 'review-fix.source.receipt.txt'
Copy-Item -LiteralPath $PSCommandPath -Destination $sourceCopy
if ((Get-FileHash -LiteralPath $sourceCopy).Hash -cne (Get-FileHash -LiteralPath $PSCommandPath).Hash) {throw 'Matrix correction source copy differs.'}
Write-Output "PASS matrix metadata: six unique descriptors, $($rows.Count) exact retained copy bindings, six digest-valid captured states; original packaging error preserved."
