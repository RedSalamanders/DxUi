// Scratch timing of the option for SetRange's sort (never part of the library): what std::sort costs on the ids of a scattered
// Ctrl+A, and what a least-significant-digit radix sort of the same ids costs. The ids are those of the grid-selection harness's
// ScrambledRowsModel (row * 2654435761 modulo 2^32), in row order, which is what SetRange copies and then has to sort for IsSelected.
// The radix sort takes one scratch buffer the size of the ids and skips a pass whose byte is the same in every id.
#include <algorithm>
#include <array>
#include <chrono>
#include <cstdint>
#include <cstdio>
#include <cstdlib>
#include <vector>

namespace
{
using Clock = std::chrono::steady_clock;

[[nodiscard]] double Milliseconds(Clock::time_point start)
{
    return std::chrono::duration<double, std::milli>(Clock::now() - start).count();
}

void RadixSort(std::vector<uint64_t>& ids, std::vector<uint64_t>& scratch)
{
    scratch.resize(ids.size());
    std::array<size_t, 256> counts{};
    uint64_t* from = ids.data();
    uint64_t* to   = scratch.data();
    for (unsigned shift = 0; shift < 64; shift += 8)
    {
        counts.fill(0);
        for (size_t index = 0; index < ids.size(); ++index)
            ++counts[(from[index] >> shift) & 0xFFu];
        if (counts[(from[0] >> shift) & 0xFFu] == ids.size())
            continue; // every id has the same byte here
        size_t offset = 0;
        for (size_t digit = 0; digit < 256; ++digit)
        {
            const size_t count = counts[digit];
            counts[digit]      = offset;
            offset += count;
        }
        for (size_t index = 0; index < ids.size(); ++index)
            to[counts[(from[index] >> shift) & 0xFFu]++] = from[index];
        std::swap(from, to);
    }
    if (from != ids.data())
        std::copy(from, from + ids.size(), ids.data());
}

[[nodiscard]] double Median(std::vector<double> values)
{
    std::sort(values.begin(), values.end());
    return values[values.size() / 2];
}
} // namespace

int main()
{
    std::printf("rows        std::sort ms   radix sort ms   same result\n");
    for (const size_t rows : {size_t{1'000}, size_t{20'000}, size_t{200'000}, size_t{1'000'000}})
    {
        std::vector<uint64_t> original(rows);
        for (size_t row = 0; row < rows; ++row)
            original[row] = static_cast<uint32_t>(static_cast<uint32_t>(row) * 2654435761u);
        std::vector<uint64_t> scratch;
        std::vector<double> standard;
        std::vector<double> radix;
        bool same = true;
        for (int repeat = 0; repeat < 21; ++repeat)
        {
            std::vector<uint64_t> a(original);
            auto start = Clock::now();
            std::sort(a.begin(), a.end());
            standard.push_back(Milliseconds(start));
            std::vector<uint64_t> b(original);
            start = Clock::now();
            RadixSort(b, scratch);
            radix.push_back(Milliseconds(start));
            same = same && a == b;
        }
        std::printf("%-10zu  %12.4f   %13.4f   %s\n", rows, Median(standard), Median(radix), same ? "yes" : "NO");
    }
    return 0;
}
